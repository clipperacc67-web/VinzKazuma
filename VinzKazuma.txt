-- Fly direction (Mode Flat Look Camera)
local function flyDir()
    if not Hum then return Vector3.zero end
    local md = Hum.MoveDirection
    if md.Magnitude < 0.05 then return Vector3.zero end
    local look = Cam.CFrame.LookVector
    local right = Cam.CFrame.RightVector
    local flat = Vector3.new(look.X, 0, look.Z)
    if flat.Magnitude > 0.01 then flat = flat.Unit else flat = Vector3.new(0,0,-1) end
    local f = md:Dot(flat)
    local r = md:Dot(right)
    local d = look * f + right * r
    if d.Magnitude > 0.01 then return d.Unit end
    return Vector3.zero
end

-- Fly Handler
function startFly()
    getChar()
    if not Root or not Hum then return end
    stopFly()

    BV = Instance.new("BodyVelocity")
    BV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    BV.Velocity = Vector3.zero
    BV.Parent = Root

    BG = Instance.new("BodyGyro")
    BG.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
    BG.P = 300000
    BG.D = 500
    BG.Parent = Root

    Hum.PlatformStand = true
    Hum.AutoRotate = false

    FlyConn = RunService.Heartbeat:Connect(function()
        if not Flying or not BV or not BV.Parent then
            stopFly()
            return
        end
        BV.Velocity = flyDir() * FlySpeed
        local look = Cam.CFrame.LookVector
        local flat = Vector3.new(look.X, 0, look.Z)
        if flat.Magnitude > 0.05 then
            BG.CFrame = CFrame.new(Root.Position, Root.Position + flat)
        end
    end)
end

function stopFly()
    if FlyConn then FlyConn:Disconnect() FlyConn = nil end
    if BV then pcall(function() BV:Destroy() end) BV = nil end
    if BG then pcall(function() BG:Destroy() end) BG = nil end
    if Hum then
        Hum.PlatformStand = false
        Hum.AutoRotate = true
    end
end

-- Noclip
function startClip()
    getChar()
    if not Char then return end
    stopClip()
    ClipConn = RunService.Stepped:Connect(function()
        if not Noclip or not Char then stopClip() return end
        for _, v in pairs(Char:GetDescendants()) do
            if v:IsA("BasePart") then v.CanCollide = false end
        end
    end)
end

function stopClip()
    if ClipConn then ClipConn:Disconnect() ClipConn = nil end
    if Char then
        for _, v in pairs(Char:GetDescendants()) do
            if v:IsA("BasePart") then v.CanCollide = true end
        end
    end
end

-- Speed
local function setSpeed()
    getChar()
    if Hum then
        if SpeedOn then
            Hum.WalkSpeed = RunSpeed
        else
            Hum.WalkSpeed = OldSpeed
        end
    end
end

-- Immune
function startImm()
    getChar()
    if not Hum then return end
    stopImm()
    pcall(function()
        Hum.MaxHealth = 9e9
        Hum.Health = 9e9
    end)
    ImmConn = RunService.Heartbeat:Connect(function()
        if not Immune or not Hum then stopImm() return end
        if Hum.Health < 9e8 then
            Hum.Health = 9e9
        end
    end)
end

function stopImm()
    if ImmConn then ImmConn:Disconnect() ImmConn = nil end
    if Hum then
        pcall(function()
            Hum.MaxHealth = 100
            Hum.Health = 100
        end)
    end
end

-- Hitbox Expander
local function startHitbox()
    HitboxConn = RunService.RenderStepped:Connect(function()
        if not HitboxOn then return end
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= LP and p.Character then
                local hrp = p.Character:FindFirstChild("HumanoidRootPart")
                local hum = p.Character:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    hrp.Size = Vector3.new(HitboxSize, HitboxSize, HitboxSize)
                    hrp.Transparency = HitboxTransparency
                    hrp.Color = Color3.fromRGB(255, 255, 255)
                    hrp.Material = Enum.Material.Neon
                    hrp.CanCollide = false
                end
            end
        end
    end)
end

local function stopHitbox()
    if HitboxConn then HitboxConn:Disconnect() HitboxConn = nil end
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LP and p.Character then
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                hrp.Size = Vector3.new(2, 2, 1)
                hrp.Transparency = 1
                hrp.CanCollide = false
            end
        end
    end
end

local ESPHold = Instance.new("Folder")
ESPHold.Name = "VinzESP"

local function clearESP()
    for _, v in pairs(ESPHold:GetChildren()) do
        v:Destroy()
    end
end

local function addESP(plr)
    if plr == LP then return end
    local c = plr.Character
    if not c then return end
    local h = c:FindFirstChild("Head") or c:FindFirstChild("HumanoidRootPart")
    if not h then return end

    local old = ESPHold:FindFirstChild(plr.Name)
    if old then old:Destroy() end

    local b = Instance.new("BillboardGui")
    b.Name = plr.Name
    b.Adornee = h
    b.Size = UDim2.new(0, 130, 0, 26)
    b.StudsOffset = Vector3.new(0, 2.6, 0)
    b.AlwaysOnTop = true
    b.MaxDistance = math.huge -- Tanpa batas jarak render engine
    b.Parent = ESPHold

    local t = Instance.new("TextLabel")
    t.Size = UDim2.new(1, 0, 1, 0)
    t.BackgroundTransparency = 1
    t.TextColor3 = Color3.fromRGB(0, 220, 255)
    t.TextStrokeTransparency = 0.3
    t.TextSize = 13
    t.Font = Enum.Font.GothamBold
    t.Parent = b

    if plr.DisplayName ~= plr.Name then
        t.Text = plr.DisplayName .. " (@" .. plr.Name .. ")"
    else
        t.Text = plr.Name
    end
end

function startESP()
    clearESP()
    local success, parent = pcall(function() return gethui() end)
    ESPHold.Parent = (success and parent) or CoreGui
    
    -- Loop global untuk memastikan player sejauh apapun langsung terdeteksi
    RunService.RenderStepped:Connect(function()
        if not ESPOn then return end
        for _, p in pairs(Players:GetPlayers()) do
            if p ~= LP and p.Character and not ESPHold:FindFirstChild(p.Name) then
                addESP(p)
            end
        end
    end)

    for _, p in pairs(Players:GetPlayers()) do
        addESP(p)
        p.CharacterAdded:Connect(function()
            if ESPOn then
                task.wait(0.5)
                addESP(p)
            end
        end)
    end
    Players.PlayerAdded:Connect(function(p)
        p.CharacterAdded:Connect(function()
            if ESPOn then
                task.wait(0.5)
                addESP(p)
            end
        end)
    end)
end

function stopESP()
    clearESP()
end

-- Teleport Global Tanpa Batas Jarak
local function tpTo(name)
    if not name or name == "" then return false end
    name = string.lower(name)
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= LP then
            if string.find(string.lower(p.Name), name, 1, true) or string.find(string.lower(p.DisplayName), name, 1, true) then
                local tr = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
                getChar()
                if tr and Root then
                    Root.CFrame = tr.CFrame + Vector3.new(0, 4, 0)
                    return true, p.Name
                end
            end
        end
    end
    return false
end

-- Fungsi Drag
local function makeDraggable(item, handle)
    handle = handle or item
    local dragging = false
    local dragInput, dragStart, startPos
    local moved = false

    handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = item.Position
            moved = false
            
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    UIS.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            if delta.Magnitude > 5 then
                moved = true
            end
            item.Position = UDim2.new(
                startPos.X.Scale, 
                startPos.X.Offset + delta.X, 
                startPos.Y.Scale, 
                startPos.Y.Offset + delta.Y
            )
        end
    end)
    
    return function()
        return moved
    end
end
