local function makeDraggable(handle, target, onClick)
    local dragging, moved, startInput, startPos = false, false, nil, nil
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging, moved = true, false
            startInput = i.Position
            startPos = target.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if not moved and onClick then onClick() end
                end
            end)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseMovement) then
            local d = i.Position - startInput
            if d.Magnitude > 6 then moved = true end
            if moved then
                target.Position = UDim2.new(
                    startPos.X.Scale, startPos.X.Offset + d.X,
                    startPos.Y.Scale, startPos.Y.Offset + d.Y
                )
            end
        end
    end)
end

local function toggleMenu(visible)
    if visible then
        mainFrame.Visible = true
        mainScale.Scale = 0.85
        TweenService:Create(mainScale, TweenInfo.new(0.25, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = 1}):Play()
    else
        local tw = TweenService:Create(mainScale, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {Scale = 0.85})
        tw:Play()
        tw.Completed:Connect(function()
            if not mainFrame.Visible or mainScale.Scale <= 0.86 then
                mainFrame.Visible = false
            end
        end)
    end
end

closeBtn.Activated:Connect(function()

end)

-- Hitbox Expander & Visual
local hitboxOn = false
local hitboxSize = 15
local hitboxAlpha = 5
local hitboxLoop = nil
local defaultProperties = {}

local function cleanHitboxes()
    for _, p in ipairs(Players:GetPlayers()) do
        if defaultProperties[p] and p.Character then
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                hrp.Size = defaultProperties[p].Size
                hrp.Transparency = defaultProperties[p].Transparency
                hrp.CanCollide = defaultProperties[p].CanCollide
            end
            local box = p.Character:FindFirstChild("VK_HitboxAdornment")
            if box then box:Destroy() end
        end
        defaultProperties[p] = nil
    end
end

local function applyHitbox()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp and p.Character then
            local char = p.Character
            local hrp = char:FindFirstChild("HumanoidRootPart")
            local hum = char:FindFirstChildOfClass("Humanoid")

            if hrp and hum and hum.Health > 0 then
                if not defaultProperties[p] then
                    defaultProperties[p] = {
                        Size = hrp.Size,
                        Transparency = hrp.Transparency,
                        CanCollide = hrp.CanCollide
                    }
                end

                hrp.Size = Vector3.new(hitboxSize, hitboxSize, hitboxSize)
                hrp.CanCollide = false
                hrp.Transparency = 1

                local box = char:FindFirstChild("VK_HitboxAdornment")
                if not box then
                    box = Instance.new("BoxHandleAdornment")
                    box.Name = "VK_HitboxAdornment"
                    box.Adornee = hrp
                    box.AlwaysOnTop = true
                    box.ZIndex = 5
                    box.Color3 = C_WHITE
                    box.Parent = char
                end

                box.Size = Vector3.new(hitboxSize, hitboxSize, hitboxSize)
                if hitboxAlpha <= 0 then
                    box.Visible = false
                    box.Transparency = 1
                else
                    box.Visible = true
                    box.Transparency = math.clamp(1 - (hitboxAlpha / 10) * 0.85, 0.15, 0.9)
                end
            end
        end
    end
end

tabPlayer:AddToggle("Hitbox Expander", false, function(on)
    hitboxOn = on
    if on then
        hitboxLoop = RunService.RenderStepped:Connect(applyHitbox)
    else
        if hitboxLoop then hitboxLoop:Disconnect(); hitboxLoop = nil end
        cleanHitboxes()
    end
end)

tabPlayer:AddStepper("Hitbox Size", 2, 60, 2, 15, function(v) hitboxSize = v end)
tabPlayer:AddStepper("Hitbox Transparan (0 = Hilang)", 0, 10, 1, 5, function(v) hitboxAlpha = v end)

                if myRoot then
                    myRoot.CFrame = p.Character.HumanoidRootPart.CFrame * CFrame.new(0, 0, 3)

                end
            end
        end
    end
end)

-- ESP Players
local espOn = false
local espData = {}
local function clearESP()
    for _, d in pairs(espData) do if d.folder then d.folder:Destroy() end end
    table.clear(espData)
end

local function updateESP()
    for p, d in pairs(espData) do
        if not p.Parent or d.char ~= p.Character then
            if d.folder then d.folder:Destroy() end
            espData[p] = nil
        end
    end
    for _, p in ipairs(Players:GetPlayers()) do
        local char = p.Character
        if p ~= lp and char and not espData[p] then
            local f = Instance.new("Folder")
            f.Name = "ESP_" .. p.Name
            f.Parent = gui

            local hl = Instance.new("Highlight")
            hl.Adornee = char
            hl.FillColor = C_ACCENT
            hl.FillTransparency = 0.65
            hl.OutlineColor = C_WHITE
            hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            hl.Parent = f

            local head = char:FindFirstChild("Head")
            if head then
                local bb = Instance.new("BillboardGui")
                bb.Adornee = head
                bb.AlwaysOnTop = true
                bb.Size = UDim2.fromOffset(120, 20)
                bb.StudsOffset = Vector3.new(0, 2.5, 0)
                bb.Parent = f

                local t = Instance.new("TextLabel")
                t.Size = UDim2.fromScale(1, 1)
                t.BackgroundTransparency = 1
                t.Text = p.DisplayName
                t.TextColor3 = C_WHITE
                t.TextStrokeTransparency = 0.2
                t.Font = Enum.Font.GothamBold
                t.TextSize = 12
                t.Parent = bb
            end
            espData[p] = { folder = f, char = char }
        end
    end
end

tabPlayer:AddToggle("Player ESP", false, function(on)
    espOn = on
    if on then
        task.spawn(function()
            while espOn do
                pcall(updateESP)
                task.wait(1)
            end
        end)
    else
        clearESP()
    end
end)

-- Smooth Mobile Fly
local flyOn, flySpeed, vertical = false, 50, 0
local flyBV, flyBG, flyConn
local currentVel = Vector3.zero

local function stopFly()
    if flyConn then flyConn:Disconnect(); flyConn = nil end
    if flyBV then flyBV:Destroy(); flyBV = nil end
    if flyBG then flyBG:Destroy(); flyBG = nil end
    currentVel = Vector3.zero
    local h = getHum()
    if h then h.AutoRotate = true end
end

local function startFly()
    stopFly()
    local hrp, hum = getRoot(), getHum()
    if not hrp or not hum then return end

    flyBV = Instance.new("BodyVelocity")
    flyBV.MaxForce = Vector3.new(1e8, 1e8, 1e8)
    flyBV.Velocity = Vector3.zero
    flyBV.Parent = hrp

    flyBG = Instance.new("BodyGyro")
    flyBG.MaxTorque = Vector3.new(1e8, 1e8, 1e8)
    flyBG.P = 15000
    flyBG.D = 800
    flyBG.CFrame = hrp.CFrame
    flyBG.Parent = hrp

    hum.AutoRotate = false

    flyConn = RunService.RenderStepped:Connect(function(dt)
        local cam = workspace.CurrentCamera
        if not cam or not hrp.Parent then return end

        local md = hum.MoveDirection
        local targetVelocity = Vector3.zero

        if md.Magnitude > 0 then
            local look = cam.CFrame.LookVector
            local flat = Vector3.new(look.X, 0, look.Z).Unit
            local camRight = cam.CFrame.RightVector
            local flatRight = Vector3.new(camRight.X, 0, camRight.Z).Unit

            local forwardPush = md:Dot(flat)
            local strafePush  = md:Dot(flatRight)

            local moveVec = (cam.CFrame.LookVector * forwardPush) + (cam.CFrame.RightVector * strafePush)
            targetVelocity = moveVec.Unit * flySpeed
        end

        targetVelocity += Vector3.new(0, vertical * flySpeed, 0)
        local smoothFactor = math.clamp(dt * 14, 0.05, 1)
        currentVel = currentVel:Lerp(targetVelocity, smoothFactor)
        flyBV.Velocity = currentVel

        local camLook = cam.CFrame.LookVector
        local faceDir = Vector3.new(camLook.X, 0, camLook.Z)
        if faceDir.Magnitude > 0.01 then
            flyBG.CFrame = flyBG.CFrame:Lerp(CFrame.new(hrp.Position, hrp.Position + faceDir), smoothFactor)
        end
    end)
end

-- Mobile Up/Down Pad
local upDown = Instance.new("Frame")
upDown.Size = UDim2.fromOffset(48, 106)
upDown.Position = UDim2.new(1, -66, 0.5, -53)
upDown.BackgroundTransparency = 1
upDown.Visible = false
upDown.Parent = gui

local function holdBtn(txt, y, dir)
    local b = Instance.new("TextButton")
    b.Size = UDim2.fromOffset(46, 46)
    b.Position = UDim2.fromOffset(0, y)
    b.BackgroundColor3 = C_ACCENT
    b.BackgroundTransparency = 0.3
    b.Text = txt
    b.TextColor3 = C_WHITE
    b.Font = Enum.Font.GothamBold
    b.TextSize = 20
    b.Parent = upDown
    addCorner(b, 23)

    b.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseButton1 then
            vertical = dir
            local conn
            conn = i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then
                    if vertical == dir then vertical = 0 end
                    conn:Disconnect()
                end
            end)
        end
    end)
end
holdBtn("▲", 0, 1)
holdBtn("▼", 54, -1)

tabMovement:AddToggle("Smooth Fly", false, function(on)
    flyOn = on
    upDown.Visible = on
    vertical = 0
    if on then startFly() else stopFly() end
end)
tabMovement:AddStepper("Fly Speed", 10, 250, 10, 50, function(v) flySpeed = v end)

local tpTool = nil
tabVisuals:AddToggle("Click TP Tool", false, function(on)
    if on then
        if not lp.Backpack:FindFirstChild("Click TP") then
            tpTool = Instance.new("Tool")
            tpTool.Name = "Click TP"
            tpTool.RequiresHandle = false
            tpTool.Activated:Connect(function()
                local mouse = lp:GetMouse()
                local hrp = getRoot()
                if hrp and mouse.Hit then
                    hrp.CFrame = CFrame.new(mouse.Hit.Position + Vector3.new(0, 3, 0))
                end
            end)
            tpTool.Parent = lp.Backpack
        end
    else
        local t = lp.Backpack:FindFirstChild("Click TP") or (lp.Character and lp.Character:FindFirstChild("Click TP"))
        if t then t:Destroy() end
    end
end)

local tabMisc = createTab("Misc")

tabMisc:AddButton("FPS Booster (Reduce Lag)", function()
    pcall(function() settings().Rendering.QualityLevel = Enum.QualityLevel.Level01 end)
    Lighting.GlobalShadows = false
    origLight.GlobalShadows = false
    for _, v in ipairs(workspace:GetDescendants()) do
        if v:IsA("BasePart") then
            v.Material = Enum.Material.SmoothPlastic
            v.CastShadow = false
        elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Smoke") or v:IsA("Fire") or v:IsA("Sparkles") then
            v.Enabled = false
        end
    end
end)

        afkConn = lp.Idled:Connect(function()

tabMisc:AddButton("Teleport ke Player Acak", function()
    local list = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            table.insert(list, p.Character.HumanoidRootPart)
        end
    end
    if #list > 0 then
        local chosen = list[math.random(1, #list)]
        local hrp = getRoot()
        if hrp then
            hrp.CFrame = chosen.CFrame * CFrame.new(0, 2, 3)
        end
    end
end)

    if h then h.Health = 0 end
end)
