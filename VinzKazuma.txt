local function makeDraggable(handle, target, onClick)
    local dragging, moved, startInput, startPos = false, false, nil, nil
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging, moved = true, false
            startInput = i.Position
            startPos = target.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if not moved and onClick then onClick() end
                end
            end)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseMovement) then
            local d = i.Position - startInput
            if d.Magnitude > 6 then moved = true end
            if moved then
                target.Position = UDim2.new(
                    startPos.X.Scale, startPos.X.Offset + d.X,
                    startPos.Y.Scale, startPos.Y.Offset + d.Y
                )
            end
        end
    end)
end

-- SILENT AIM ENGINE (METAMETHOD HOOK)

local silentAimOn = false

local function getClosestEnemyTarget()
    local closest, minDist = nil, math.huge
    local myRoot = getRoot()
    if not myRoot then return nil end

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp and p.Character then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            local head = p.Character:FindFirstChild("Head")
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if hum and hum.Health > 0 and (head or hrp) then
                local targetPart = head or hrp
                local dist = (targetPart.Position - myRoot.Position).Magnitude
                if dist < minDist then
                    minDist = dist
                    closest = targetPart
                end
            end
        end
    end
    return closest
end

pcall(function()
    local gmt = getrawmetatable and getrawmetatable(game)
    if gmt and setreadonly then
        setreadonly(gmt, false)
        local oldIndex = gmt.__index
        local oldNamecall = gmt.__namecall

        gmt.__index = newcclosure(function(t, k)
            if silentAimOn and not checkcaller() and t:IsA("Mouse") and (k == "Hit" or k == "Target") then
                local target = getClosestEnemyTarget()
                if target then
                    if k == "Hit" then return target.CFrame
                    elseif k == "Target" then return target end
                end
            end
            return oldIndex(t, k)
        end)

        gmt.__namecall = newcclosure(function(self, ...)
            local method = getnamecallmethod()
            local args = {...}
            if silentAimOn and not checkcaller() then
                if method == "Raycast" and self == workspace then
                    local target = getClosestEnemyTarget()
                    if target then
                        local origin = args[1]
                        args[2] = (target.Position - origin).Unit * 1000
                        return oldNamecall(self, unpack(args))
                    end
                elseif (method == "FindPartOnRay" or method == "FindPartOnRayWithWhitelist" or method == "FindPartOnRayWithIgnoreList") and self == workspace then
                    local target = getClosestEnemyTarget()
                    if target and typeof(args[1]) == "Ray" then
                        local origin = args[1].Origin
                        args[1] = Ray.new(origin, (target.Position - origin).Unit * 1000)
                        return oldNamecall(self, unpack(args))
                    end
                end
            end
            return oldNamecall(self, ...)
        end)
        setreadonly(gmt, true)
    end
end)

local immuneOn = false
local immuneConn = nil
local function setupImmune()
    local hum = getHum()
    if hum then
        if immuneConn then immuneConn:Disconnect() end
        immuneConn = hum.HealthChanged:Connect(function(newHealth)
            if immuneOn and newHealth < hum.MaxHealth then
                hum.Health = hum.MaxHealth
            end
        end)
        if immuneOn and hum.Health < hum.MaxHealth then
            hum.Health = hum.MaxHealth
        end
    end
end

    immuneOn = on
    if on then setupImmune()
    else
        if immuneConn then immuneConn:Disconnect(); immuneConn = nil end
    end
end)

    if immuneOn then
        task.wait(0.5)

    end
end)

-- 4. Hitbox Expander & Box Visual (Mulai dari 0)
local hitboxOn = false
local hitboxSize = 15
local hitboxAlpha = 0
local hitboxLoop = nil
local defaultProperties = {}

local function cleanHitboxes()
    for _, p in ipairs(Players:GetPlayers()) do
        if defaultProperties[p] and p.Character then
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                hrp.Size = defaultProperties[p].Size
                hrp.Transparency = defaultProperties[p].Transparency
                hrp.CanCollide = defaultProperties[p].CanCollide
            end
            local box = p.Character:FindFirstChild("VK_HitboxAdornment")
            if box then box:Destroy() end
        end
        defaultProperties[p] = nil
    end
end

local function applyHitbox()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp and p.Character then
            local char = p.Character
            local hrp = char:FindFirstChild("HumanoidRootPart")
            local hum = char:FindFirstChildOfClass("Humanoid")

            if hrp and hum and hum.Health > 0 then
                if not defaultProperties[p] then
                    defaultProperties[p] = {
                        Size = hrp.Size,
                        Transparency = hrp.Transparency,
                        CanCollide = hrp.CanCollide
                    }
                end

                hrp.Size = Vector3.new(hitboxSize, hitboxSize, hitboxSize)
                hrp.CanCollide = false
                hrp.Transparency = 1

                local box = char:FindFirstChild("VK_HitboxAdornment")
                if not box then
                    box = Instance.new("BoxHandleAdornment")
                    box.Name = "VK_HitboxAdornment"
                    box.Adornee = hrp
                    box.AlwaysOnTop = true
                    box.ZIndex = 5
                    box.Color3 = C_WHITE
                    box.Parent = char
                end

                box.Size = Vector3.new(hitboxSize, hitboxSize, hitboxSize)
                if hitboxAlpha <= 0 then
                    box.Visible = false
                    box.Transparency = 1
                else
                    box.Visible = true
                    box.Transparency = math.clamp(1 - (hitboxAlpha / 10) * 0.85, 0.15, 0.95)
                end
            end
        end
    end
end

tabPlayer:AddToggle("Hitbox Expander", false, function(on)
    hitboxOn = on
    if on then
        hitboxLoop = RunService.RenderStepped:Connect(applyHitbox)
    else
        if hitboxLoop then hitboxLoop:Disconnect(); hitboxLoop = nil end
        cleanHitboxes()
    end

                bb.Size = UDim2.fromOffset(120, 20)

-- Ultra Smooth Mobile Fly 3D (Mengikuti Arah Kamera 3D Penuh, Tanpa Tombol Naik/Turun)
local flyOn, flySpeed = false, 50
local flyBV, flyBG, flyConn
local currentVel = Vector3.zero

local function stopFly()
    if flyConn then flyConn:Disconnect(); flyConn = nil end
    if flyBV then flyBV:Destroy(); flyBV = nil end
    if flyBG then flyBG:Destroy(); flyBG = nil end
    currentVel = Vector3.zero
    local h = getHum()
    if h then h.AutoRotate = true end
end

local function startFly()
    stopFly()
    local hrp, hum = getRoot(), getHum()
    if not hrp or not hum then return end

    flyBV = Instance.new("BodyVelocity")
    flyBV.MaxForce = Vector3.new(1e8, 1e8, 1e8)
    flyBV.Velocity = Vector3.zero
    flyBV.Parent = hrp

    flyBG = Instance.new("BodyGyro")
    flyBG.MaxTorque = Vector3.new(1e8, 1e8, 1e8)
    flyBG.P = 15000
    flyBG.D = 800
    flyBG.CFrame = hrp.CFrame
    flyBG.Parent = hrp

    hum.AutoRotate = false

    flyConn = RunService.RenderStepped:Connect(function(dt)
        local cam = workspace.CurrentCamera
        if not cam or not hrp.Parent then return end

        local md = hum.MoveDirection
        local targetVelocity = Vector3.zero

        if md.Magnitude > 0 then
            local flatLook = Vector3.new(cam.CFrame.LookVector.X, 0, cam.CFrame.LookVector.Z)
            local flatRight = Vector3.new(cam.CFrame.RightVector.X, 0, cam.CFrame.RightVector.Z)
            
            if flatLook.Magnitude > 0.01 and flatRight.Magnitude > 0.01 then
                local forwardPush = md:Dot(flatLook.Unit)
                local strafePush  = md:Dot(flatRight.Unit)

                -- Terbang 3D: Arah hadap kamera atas/bawah otomatis membawa terbang naik/turun
                local moveVec = (cam.CFrame.LookVector * forwardPush) + (cam.CFrame.RightVector * strafePush)
                if moveVec.Magnitude > 0.05 then
                    targetVelocity = moveVec.Unit * flySpeed
                end
            end
        end

        local smoothFactor = math.clamp(dt * 12, 0.05, 1)
        currentVel = currentVel:Lerp(targetVelocity, smoothFactor)
        flyBV.Velocity = currentVel

        local faceDir = cam.CFrame.LookVector
        if faceDir.Magnitude > 0.01 then
            flyBG.CFrame = flyBG.CFrame:Lerp(CFrame.new(hrp.Position, hrp.Position + faceDir), smoothFactor)
        end
    end)
end

tabMovement:AddToggle("Ultra Smooth Fly (3D Cam)", false, function(on)
    flyOn = on
    if on then startFly() else stopFly() end
end)
tabMovement:AddStepper("Fly Speed", 10, 250, 10, 50, function(v) flySpeed = v end)

-- Free Camera Mobile (Filter Touch Zona Analog agar Tidak Senggol)
local freecamOn = false
local freecamConn = nil
local freecamTouchConn = nil
local freecamPos = Vector3.zero
local freecamYaw, freecamPitch = 0, 0
local freecamSpeed = 45

local function isInAnalogDeadzone(pos)
    local cam = workspace.CurrentCamera
    if not cam then return false end
    local viewport = cam.ViewportSize
    -- Zona analog mobile: sisi kiri bawah (lebar < 45% dan tinggi > 40%)
    return pos.X < viewport.X * 0.45 and pos.Y > viewport.Y * 0.40
end

tabVisuals:AddToggle("Freecam (Analog & Touch)", false, function(on)
    freecamOn = on
    local cam = workspace.CurrentCamera
    local hrp = getRoot()
    local hum = getHum()
    if not cam then return end

    if on then
        if hrp then hrp.Anchored = true end
        freecamPos = cam.CFrame.Position
        local rx, ry, _ = cam.CFrame:ToOrientation()
        freecamPitch = rx
        freecamYaw = ry

        cam.CameraType = Enum.CameraType.Scriptable

        -- Usapan layar memutar sudut kamera (Sentuhan di zona analog diabaikan)
        freecamTouchConn = UIS.InputChanged:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement then
                if input.UserInputType == Enum.UserInputType.Touch and isInAnalogDeadzone(input.Position) then
                    return
                end
                local delta = input.Delta
                freecamYaw = freecamYaw - math.rad(delta.X * 0.35)
                freecamPitch = math.clamp(freecamPitch - math.rad(delta.Y * 0.35), math.rad(-89), math.rad(89))
            end
        end)

        -- Analog bawaan menggerakkan posisi kamera
        freecamConn = RunService.RenderStepped:Connect(function(dt)
            if freecamOn then
                local currentRot = CFrame.Angles(0, freecamYaw, 0) * CFrame.Angles(freecamPitch, 0, 0)
                local moveDir = hum and hum.MoveDirection or Vector3.zero
                if moveDir.Magnitude > 0 then
                    local forwardPush = moveDir:Dot(cam.CFrame.LookVector)
                    local strafePush  = moveDir:Dot(cam.CFrame.RightVector)
                    local camMove = (currentRot.LookVector * forwardPush) + (currentRot.RightVector * strafePush)
                    freecamPos = freecamPos + camMove.Unit * (freecamSpeed * dt)
                end
                cam.CFrame = CFrame.new(freecamPos) * currentRot
            end
        end)
    else
        if freecamConn then freecamConn:Disconnect(); freecamConn = nil end
        if freecamTouchConn then freecamTouchConn:Disconnect(); freecamTouchConn = nil end
        if hrp then hrp.Anchored = false end
        cam.CameraType = Enum.CameraType.Custom
        if hum then cam.CameraSubject = hum end
    end
end)

-- Server Hop Engine
local function serverHop(lower)
    local placeId = game.PlaceId
    local currentJob = game.JobId
    local url = string.format("https://games.roblox.com/v1/games/%s/servers/Public?sortOrder=%s&limit=100", tostring(placeId), lower and "Asc" or "Desc")
    
    local success, response = pcall(function()
        return game:HttpGet(url)
    end)

    if success and response then
        local data = HttpService:JSONDecode(response)
        if data and data.data then
            local validServers = {}
            for _, s in ipairs(data.data) do
                if s.id ~= currentJob and s.playing and s.playing < s.maxPlayers and s.playing > 0 then
                    table.insert(validServers, s)
                end
            end

            if #validServers > 0 then
                if lower then
                    table.sort(validServers, function(a, b) return a.playing < b.playing end)
                    TeleportService:TeleportToPlaceInstance(placeId, validServers[1].id, lp)
                else
                    local picked = validServers[math.random(1, #validServers)]
                    TeleportService:TeleportToPlaceInstance(placeId, picked.id, lp)
                end
                return
            end
        end
    end
    TeleportService:Teleport(placeId, lp)
end

tabMisc:AddButton("Change Server (Random)", function()
    serverHop(false)
end)

tabMisc:AddButton("Change Lower Server (Sepi)", function()
    serverHop(true)
end)
