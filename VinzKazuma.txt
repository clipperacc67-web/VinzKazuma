local function makeDraggable(handle, target, onClick)
    local dragging, moved, startInput, startPos = false, false, nil, nil
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging, moved = true, false
            startInput = i.Position
            startPos = target.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if not moved and onClick then onClick() end
                end
            end)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseMovement) then
            local d = i.Position - startInput
            if d.Magnitude > 6 then moved = true end
            if moved then
                target.Position = UDim2.new(
                    startPos.X.Scale, startPos.X.Offset + d.X,
                    startPos.Y.Scale, startPos.Y.Offset + d.Y
                )
            end
        end
    end)
end

local stepOrder = 0
local function makeStepper(name, min, max, step, default, onChange)
    stepOrder += 1
    local value = default
    local row = Instance.new("Frame")
    row.LayoutOrder = stepOrder
    row.Size = UDim2.new(1, -4, 0, 26)
    row.BackgroundColor3 = OFF
    row.Parent = stepScroll
    corner(row, 6)

    local lbl = Instance.new("TextLabel")
    lbl.Size = UDim2.new(0.55, 0, 1, 0)
    lbl.Position = UDim2.fromOffset(8, 0)
    lbl.BackgroundTransparency = 1
    lbl.TextColor3 = WHITE
    lbl.Font = Enum.Font.GothamMedium
    lbl.TextSize = 11
    lbl.TextXAlignment = Enum.TextXAlignment.Left
    lbl.Parent = row

    local function refresh()
        lbl.Text = string.format("%s: %s", name, tostring(value))
        onChange(value)
    end

    local function btn(txt, xScale)
        local b = Instance.new("TextButton")
        b.Size = UDim2.new(0.18, 0, 0, 20)
        b.Position = UDim2.new(xScale, 0, 0.5, -10)
        b.BackgroundColor3 = ACCENT
        b.Text = txt
        b.TextColor3 = WHITE
        b.Font = Enum.Font.GothamBold
        b.TextSize = 13
        b.Parent = row
        corner(b, 5)
        return b
    end

    btn("-", 0.60).Activated:Connect(function()
        value = math.clamp(value - step, min, max)
        refresh()
    end)
    btn("+", 0.80).Activated:Connect(function()
        value = math.clamp(value + step, min, max)
        refresh()
    end)
    refresh()
end

-- SMOOTH FLY ENGINE (Interpolasi Halus untuk Android/Touch)
---
local flyOn, vertical = false, 0
local flyBV, flyBG, flyConn
local currentVel = Vector3.zero

local function stopFly()
    if flyConn then flyConn:Disconnect(); flyConn = nil end
    if flyBV then flyBV:Destroy(); flyBV = nil end
    if flyBG then flyBG:Destroy(); flyBG = nil end
    currentVel = Vector3.zero
    local h = getHum()
    if h then h.AutoRotate = true end
end

local function startFly()
    stopFly()
    local hrp, hum = getRoot(), getHum()
    if not hrp or not hum then return end

    flyBV = Instance.new("BodyVelocity")
    flyBV.MaxForce = Vector3.new(1e8, 1e8, 1e8)
    flyBV.Velocity = Vector3.zero
    flyBV.Parent = hrp

    flyBG = Instance.new("BodyGyro")
    flyBG.MaxTorque = Vector3.new(1e8, 1e8, 1e8)
    flyBG.P = 15000
    flyBG.D = 800
    flyBG.CFrame = hrp.CFrame
    flyBG.Parent = hrp

    hum.AutoRotate = false

    flyConn = RunService.RenderStepped:Connect(function(dt)
        local cam = workspace.CurrentCamera
        if not cam or not hrp.Parent then return end

        local md = hum.MoveDirection
        local targetVelocity = Vector3.zero

        if md.Magnitude > 0 then
            local look = cam.CFrame.LookVector
            local flat = Vector3.new(look.X, 0, look.Z).Unit
            local camRight = cam.CFrame.RightVector
            local flatRight = Vector3.new(camRight.X, 0, camRight.Z).Unit

            local forwardPush = md:Dot(flat)
            local strafePush  = md:Dot(flatRight)

            local moveVec = (cam.CFrame.LookVector * forwardPush) + (cam.CFrame.RightVector * strafePush)
            targetVelocity = moveVec.Unit * flySpeed
        end

        targetVelocity += Vector3.new(0, vertical * flySpeed, 0)

        -- Smooth Lerp (mengurangi stuttering saat joystick berbelok)
        local smoothFactor = math.clamp(dt * 14, 0.05, 1)
        currentVel = currentVel:Lerp(targetVelocity, smoothFactor)
        flyBV.Velocity = currentVel

        local camLook = cam.CFrame.LookVector
        local faceDir = Vector3.new(camLook.X, 0, camLook.Z)
        if faceDir.Magnitude > 0.01 then
            flyBG.CFrame = flyBG.CFrame:Lerp(CFrame.new(hrp.Position, hrp.Position + faceDir), smoothFactor)
        end
    end)
end

-- FITUR: HITBOX EXPANDER + VISUAL BOX (Enemy/Target Players Only)
---
local hitboxOn = false
local hitboxLoop = nil
local defaultProperties = {}

local function cleanHitboxForPlayer(p)
    if defaultProperties[p] then
        local char = p.Character
        if char then
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if hrp then
                hrp.Size = defaultProperties[p].Size or Vector3.new(2, 2, 1)
                hrp.Transparency = defaultProperties[p].Transparency or 1
                hrp.CanCollide = defaultProperties[p].CanCollide or false
            end
            local box = char:FindFirstChild("VK_HitboxAdornment")
            if box then box:Destroy() end
        end
        defaultProperties[p] = nil
    end
end

local function cleanAllHitboxes()
    for _, p in ipairs(Players:GetPlayers()) do
        cleanHitboxForPlayer(p)
    end
end

local function applyHitbox()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp and p.Character then
            local char = p.Character
            local hrp = char:FindFirstChild("HumanoidRootPart")
            local hum = char:FindFirstChildOfClass("Humanoid")

            if hrp and hum and hum.Health > 0 then
                -- Backup state awal
                if not defaultProperties[p] then
                    defaultProperties[p] = {
                        Size = hrp.Size,
                        Transparency = hrp.Transparency,
                        CanCollide = hrp.CanCollide
                    }
                end

                -- Ubah ukuran collision box
                hrp.Size = Vector3.new(hitboxSize, hitboxSize, hitboxSize)
                hrp.CanCollide = false
                hrp.Transparency = 1

                -- Visualisasi Kotak Putih
                local box = char:FindFirstChild("VK_HitboxAdornment")
                if not box then
                    box = Instance.new("BoxHandleAdornment")
                    box.Name = "VK_HitboxAdornment"
                    box.Adornee = hrp
                    box.AlwaysOnTop = true
                    box.ZIndex = 5
                    box.Color3 = WHITE
                    box.Parent = char
                end

                box.Size = Vector3.new(hitboxSize, hitboxSize, hitboxSize)

                -- Kontrol Alpha: Nilai 0 = Hilang (Full Transparan), 1-10 = Solid bertahap
                if hitboxAlpha <= 0 then
                    box.Visible = false
                    box.Transparency = 1
                else
                    box.Visible = true
                    -- Mapping 1 -> 0.85 (tipis), 10 -> 0.15 (jelas)
                    box.Transparency = math.clamp(1 - (hitboxAlpha / 10) * 0.85, 0.1, 0.9)
                end
            end
        end
    end
end

addToggle("Hitbox", function(on)
    hitboxOn = on
    if on then
        hitboxLoop = RunService.RenderStepped:Connect(function()
            applyHitbox()
        end)
    else
        if hitboxLoop then hitboxLoop:Disconnect(); hitboxLoop = nil end
        cleanAllHitboxes()
    end
end)

Players.PlayerRemoving:Connect(function(p)
    defaultProperties[p] = nil
end)

local espOn = false
local espData = {}

local function clearESP()
    for _, d in pairs(espData) do 
        if d.folder then d.folder:Destroy() end 
    end
    table.clear(espData)
end

local function updateESP()
    for p, d in pairs(espData) do
        if not p.Parent or d.char ~= p.Character then
            if d.folder then d.folder:Destroy() end
            espData[p] = nil
        end
    end

    for _, p in ipairs(Players:GetPlayers()) do
        local char = p.Character
        if p ~= lp and char and not espData[p] then
            local f = Instance.new("Folder")
            f.Name = "ESP_" .. p.Name
            f.Parent = gui

            local hl = Instance.new("Highlight")
            hl.Adornee = char
            hl.FillColor = ACCENT
            hl.FillTransparency = 0.65
            hl.OutlineColor = WHITE
            hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            hl.Parent = f

            local head = char:FindFirstChild("Head")
            if head then
                local bb = Instance.new("BillboardGui")
                bb.Adornee = head
                bb.AlwaysOnTop = true
                bb.Size = UDim2.fromOffset(110, 18)
                bb.StudsOffset = Vector3.new(0, 2.5, 0)
                bb.Parent = f

                local t = Instance.new("TextLabel")
                t.Size = UDim2.fromScale(1, 1)
                t.BackgroundTransparency = 1
                t.Text = p.DisplayName
                t.TextColor3 = WHITE
                t.TextStrokeTransparency = 0.2
                t.Font = Enum.Font.GothamBold
                t.TextSize = 12
                t.Parent = bb
            end
            espData[p] = { folder = f, char = char }
        end
    end
end

addToggle("ESP", function(on)
    espOn = on
    if on then
        task.spawn(function()
            while espOn do
                pcall(updateESP)
                task.wait(1)
            end
        end)
    else
        clearESP()
    end
end)
