local function addRipple(btn)
    btn.ClipsDescendants = true
    local sc = Instance.new("UIScale", btn)
    btn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            TweenService:Create(sc, TweenInfo.new(0.12, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Scale = 0.96}):Play()
            local ripple = Instance.new("Frame")
            ripple.BackgroundColor3 = C_WHITE
            ripple.BackgroundTransparency = 0.72
            ripple.BorderSizePixel = 0
            local relX = input.Position.X - btn.AbsolutePosition.X
            local relY = input.Position.Y - btn.AbsolutePosition.Y
            ripple.Position = UDim2.fromOffset(relX, relY)
            ripple.AnchorPoint = Vector2.new(0.5, 0.5)
            ripple.Size = UDim2.fromOffset(0, 0)
            ripple.ZIndex = btn.ZIndex + 2
            ripple.Parent = btn
            addCorner(ripple, 999)

            local maxD = math.max(btn.AbsoluteSize.X, btn.AbsoluteSize.Y) * 2.2
            local tw = TweenService:Create(ripple, TweenInfo.new(0.45, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {
                Size = UDim2.fromOffset(maxD, maxD),
                BackgroundTransparency = 1
            })
            tw:Play()
            tw.Completed:Connect(function() ripple:Destroy() end)
        end
    end)
    btn.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            TweenService:Create(sc, TweenInfo.new(0.2, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {Scale = 1}):Play()
        end
    end)
end

local function makeDraggable(handle, target, onClick)
    local dragging, moved, startInput, startPos = false, false, nil, nil
    handle.InputBegan:Connect(function(i)
        if i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging, moved = true, false
            startInput = i.Position
            startPos = target.Position
            i.Changed:Connect(function()
                if i.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if not moved and onClick then onClick() end
                end
            end)
        end
    end)
    UIS.InputChanged:Connect(function(i)
        if dragging and (i.UserInputType == Enum.UserInputType.Touch or i.UserInputType == Enum.UserInputType.MouseMovement) then
            local d = i.Position - startInput
            if d.Magnitude > 6 then moved = true end
            if moved then
                local goal = UDim2.new(
                    startPos.X.Scale, startPos.X.Offset + d.X,
                    startPos.Y.Scale, startPos.Y.Offset + d.Y
                )
                TweenService:Create(target, TweenInfo.new(0.08, Enum.EasingStyle.Quart, Enum.EasingDirection.Out), {Position = goal}):Play()
            end
        end
    end)
end

local silentAimOn = false

local function getClosestEnemyTarget()
    local closest, minDist = nil, math.huge
    local myRoot = getRoot()
    if not myRoot then return nil end

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp and p.Character then
            local hum = p.Character:FindFirstChildOfClass("Humanoid")
            local head = p.Character:FindFirstChild("Head")
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if hum and hum.Health > 0 and (head or hrp) then
                local targetPart = head or hrp
                local dist = (targetPart.Position - myRoot.Position).Magnitude
                if dist < minDist then
                    minDist = dist
                    closest = targetPart
                end
            end
        end
    end
    return closest
end

pcall(function()
    local gmt = getrawmetatable and getrawmetatable(game)
    if gmt and setreadonly and newcclosure then
        setreadonly(gmt, false)
        local oldIndex = gmt.__index
        local oldNamecall = gmt.__namecall

        gmt.__index = newcclosure(function(t, k)
            if silentAimOn and (not checkcaller or not checkcaller()) and t:IsA("Mouse") and (k == "Hit" or k == "Target") then
                local target = getClosestEnemyTarget()
                if target then
                    if k == "Hit" then return target.CFrame
                    elseif k == "Target" then return target end
                end
            end
            return oldIndex(t, k)
        end)

        gmt.__namecall = newcclosure(function(self, ...)
            local method = getnamecallmethod()
            local args = {...}
            if silentAimOn and (not checkcaller or not checkcaller()) then
                if method == "Raycast" and self == workspace then
                    local target = getClosestEnemyTarget()
                    if target then
                        local origin = args[1]
                        args[2] = (target.Position - origin).Unit * 1000
                        return oldNamecall(self, unpack(args))
                    end
                elseif (method == "FindPartOnRay" or method == "FindPartOnRayWithWhitelist" or method == "FindPartOnRayWithIgnoreList") and self == workspace then
                    local target = getClosestEnemyTarget()
                    if target and typeof(args[1]) == "Ray" then
                        local origin = args[1].Origin
                        args[1] = Ray.new(origin, (target.Position - origin).Unit * 1000)
                        return oldNamecall(self, unpack(args))
                    end
                end
            end
            return oldNamecall(self, ...)
        end)
        setreadonly(gmt, true)
    end
end)

local isTargetedFlinging = false
local function flingTargetPlayer(targetPlayer)
    if isTargetedFlinging or not targetPlayer or not targetPlayer.Character then return end
    local myHrp, myHum = getRoot(), getHum()
    if not myHrp or not myHum then return end

    isTargetedFlinging = true
    local savedCFrame = myHrp.CFrame
    local anchor = getOrCreateCamAnchor()
    anchor.CFrame = savedCFrame
    local cam = workspace.CurrentCamera
    if cam then cam.CameraSubject = anchor end

    myHum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
    myHum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)

    local startTime = tick()
    local movel = 0.1

    task.spawn(function()
        while isTargetedFlinging and (tick() - startTime) < 0.85 do
            RunService.Heartbeat:Wait()
            if not targetPlayer.Parent or not targetPlayer.Character then break end
            local tHrp = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
            if not tHrp or not myHrp.Parent then break end

            myHrp.CFrame = CFrame.new(tHrp.Position + (tHrp.Velocity * 0.05))
            myHrp.Velocity = Vector3.new(10000, 10000, 10000)
            myHrp.RotVelocity = Vector3.new(50000, 50000, 50000)

            RunService.RenderStepped:Wait()
            if myHrp.Parent then myHrp.Velocity = Vector3.zero; myHrp.RotVelocity = Vector3.zero end

            RunService.Stepped:Wait()
            if myHrp.Parent then myHrp.Velocity = Vector3.new(0, movel, 0); movel = -movel end
        end

        if myHrp and myHrp.Parent then
            myHrp.CFrame = savedCFrame
            myHrp.Velocity = Vector3.zero
            myHrp.RotVelocity = Vector3.zero
            pcall(function() myHrp.AssemblyLinearVelocity = Vector3.zero; myHrp.AssemblyAngularVelocity = Vector3.zero end)
        end
        if myHum and myHum.Parent then
            myHum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, true)
            myHum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, true)
            if cam then cam.CameraSubject = myHum end
        end
        isTargetedFlinging = false
    end)
end

local hitboxOn, hitboxSize, hitboxAlpha = false, 15, 0
local hitboxLoop = nil
local defaultProperties = {}

local function cleanHitboxes()
    for _, p in ipairs(Players:GetPlayers()) do
        if defaultProperties[p] and p.Character then
            local hrp = p.Character:FindFirstChild("HumanoidRootPart")
            if hrp then
                hrp.Size = defaultProperties[p].Size
                hrp.Transparency = defaultProperties[p].Transparency
                hrp.CanCollide = defaultProperties[p].CanCollide
            end
            local box = p.Character:FindFirstChild("VK_HitboxAdornment")
            if box then box:Destroy() end
        end
        defaultProperties[p] = nil
    end
end

local function applyHitbox()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp and p.Character then
            local char = p.Character
            local hrp = char:FindFirstChild("HumanoidRootPart")
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hrp and hum and hum.Health > 0 then
                if not defaultProperties[p] then
                    defaultProperties[p] = { Size = hrp.Size, Transparency = hrp.Transparency, CanCollide = hrp.CanCollide }
                end
                hrp.Size = Vector3.new(hitboxSize, hitboxSize, hitboxSize)
                hrp.CanCollide = false
                hrp.Transparency = 1
                local box = char:FindFirstChild("VK_HitboxAdornment")
                if not box then
                    box = Instance.new("BoxHandleAdornment")
                    box.Name = "VK_HitboxAdornment"
                    box.Adornee = hrp
                    box.AlwaysOnTop = true
                    box.ZIndex = 5
                    box.Color3 = C_WHITE
                    box.Parent = char
                end
                box.Size = Vector3.new(hitboxSize, hitboxSize, hitboxSize)
                if hitboxAlpha <= 0 then
                    box.Visible = false
                    box.Transparency = 1
                else
                    box.Visible = true
                    box.Transparency = math.clamp(1 - (hitboxAlpha / 10) * 0.85, 0.15, 0.95)
                end
            end
        end
    end
end

            local dist = math.floor((d.char.HumanoidRootPart.Position - myRoot.Position).Magnitude)
            d.label.Text = string.format("%s [%ds]", p.DisplayName, dist)
        end
    end
    for _, p in ipairs(Players:GetPlayers()) do
        local char = p.Character
        if p ~= lp and char and not espData[p] then
            local f = Instance.new("Folder", gui)
            f.Name = "ESP_" .. p.Name
            local hl = Instance.new("Highlight", f)
            hl.Adornee = char
            hl.FillColor = C_ACCENT
            hl.FillTransparency = 0.65
            hl.OutlineColor = C_WHITE
            hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
            local head = char:FindFirstChild("Head")
            local t = nil
            if head then
                local bb = Instance.new("BillboardGui", f)

local function stopFly()

    if flyBG then flyBG:Destroy(); flyBG = nil end
    currentVel = Vector3.zero

local function startFly()
    stopFly()
    local char, hrp, hum = lp.Character, getRoot(), getHum()
    if not hrp or not hum or not char then return end
    if char:FindFirstChild("Animate") then char.Animate.Enabled = false end
    for _, track in ipairs(hum:GetPlayingAnimationTracks()) do track:Stop(0.05) end
    hum.PlatformStand = true
    hum.AutoRotate = false

    flyBV = Instance.new("BodyVelocity", hrp)
    flyBV.MaxForce = Vector3.new(1e8, 1e8, 1e8)
    flyBV.Velocity = Vector3.zero

    flyBG = Instance.new("BodyGyro", hrp)
    flyBG.MaxTorque = Vector3.new(1e8, 1e8, 1e8)
    flyBG.P = 25000
    flyBG.D = 600
    flyBG.CFrame = hrp.CFrame

    flyConn = RunService.RenderStepped:Connect(function(dt)
        local cam = workspace.CurrentCamera
        if not cam or not hrp.Parent then return end
        local md = hum.MoveDirection
        local targetVelocity = Vector3.zero
        if md.Magnitude > 0 then
            local flatLook = Vector3.new(cam.CFrame.LookVector.X, 0, cam.CFrame.LookVector.Z)
            local flatRight = Vector3.new(cam.CFrame.RightVector.X, 0, cam.CFrame.RightVector.Z)
            if flatLook.Magnitude > 0.01 and flatRight.Magnitude > 0.01 then
                local moveVec = (cam.CFrame.LookVector * md:Dot(flatLook.Unit)) + (cam.CFrame.RightVector * md:Dot(flatRight.Unit))
                if moveVec.Magnitude > 0.05 then targetVelocity = moveVec.Unit * flySpeed end
            end
        end
        local smoothFactor = math.clamp(dt * 14, 0.05, 1)
        currentVel = currentVel:Lerp(targetVelocity, smoothFactor)
        flyBV.Velocity = currentVel
        flyBG.CFrame = flyBG.CFrame:Lerp(CFrame.new(hrp.Position, hrp.Position + cam.CFrame.LookVector), smoothFactor)
    end)
end

local function RetainPart(Part)
    if Part:IsA("BasePart") and not Part.Anchored and Part:IsDescendantOf(workspace) then
        if Part.Parent == lp.Character or Part:IsDescendantOf(lp.Character) then return false end
        if Part.Parent and (Part.Parent:FindFirstChildOfClass("Humanoid") or Part.Parent:FindFirstChild("Head")) then return false end
        pcall(function()

        return true
    end

local function addRingPart(part)
    if RetainPart(part) and not table.find(ringParts, part) then table.insert(ringParts, part) end
end
local function removeRingPart(part)
    local idx = table.find(ringParts, part)
    if idx then table.remove(ringParts, idx) end
end

RunService.Heartbeat:Connect(function(dt)
    if not ringPartsEnabled then return end
    ringTick = ringTick + dt * ringConfig.rotationSpeed
    pcall(function()
        lp.ReplicationFocus = workspace
        sethiddenproperty(lp, "SimulationRadius", math.huge)
        sethiddenproperty(lp, "MaxSimulationRadius", math.huge)
    end)

    local hrp = getRoot()
    if not hrp then return end
    local center = hrp.Position
    local flatLook = Vector3.new(hrp.CFrame.LookVector.X, 0, hrp.CFrame.LookVector.Z)
    if flatLook.Magnitude < 0.01 then flatLook = Vector3.new(0, 0, -1) end
    local baseCF = CFrame.new(center, center + flatLook.Unit)
    local spinCF = CFrame.new(center) * CFrame.Angles(0, ringTick, 0)

    local activeList = {}
    for _, p in ipairs(ringParts) do
        if p and p.Parent and not p.Anchored and (p.Position - center).Magnitude < 350 then
            table.insert(activeList, p)
        end
    end

    local total = #activeList
    if total == 0 then return end
    local R, H, mode = ringConfig.radius, ringConfig.height, ringConfig.shape

    for i, part in ipairs(activeList) do
        local frac = (i - 1) / total
        local fixedAngle = frac * math.pi * 2
        local targetPos = center

        if mode == "Love" then
            local hx = 16 * (math.sin(fixedAngle) ^ 3)
            local hy = 13 * math.cos(fixedAngle) - 5 * math.cos(2 * fixedAngle) - 2 * math.cos(3 * fixedAngle) - math.cos(4 * fixedAngle)
            local scale = R / 15
            targetPos = (spinCF * CFrame.new(Vector3.new(hx * scale, (hy * scale) + (H * 0.2), 0))).Position
        elseif mode == "Horizontal" then
            targetPos = (spinCF * CFrame.new(Vector3.new(math.cos(fixedAngle) * R, 0, math.sin(fixedAngle) * R))).Position
        elseif mode == "Vertical" then
            local rotWheel = baseCF * CFrame.Angles(0, 0, ringTick)
            targetPos = (rotWheel * CFrame.new(Vector3.new(math.cos(fixedAngle) * R, math.sin(fixedAngle) * R + (H * 0.15), 0))).Position
        elseif mode == "Infinity" then
            local denom = 1 + (math.sin(fixedAngle) ^ 2)
            local ix = (R * 1.3 * math.cos(fixedAngle)) / denom
            local iy = (R * 1.3 * math.sin(fixedAngle) * math.cos(fixedAngle)) / denom
            targetPos = (spinCF * CFrame.new(Vector3.new(ix, iy + (H * 0.15), 0))).Position
        elseif mode == "Sphere" then
            local phi = math.acos(1 - 2 * ((i - 0.5) / total))
            local theta = math.pi * (1 + math.sqrt(5)) * i
            targetPos = (spinCF * CFrame.new(Vector3.new(math.cos(theta) * math.sin(phi) * R, math.cos(phi) * R, math.sin(theta) * math.sin(phi) * R))).Position
        elseif mode == "Tornado" then
            local spireAngle = (frac * math.pi * 10)
            local curR = R * (0.2 + 0.8 * frac)
            targetPos = (spinCF * CFrame.new(Vector3.new(math.cos(spireAngle) * curR, (frac - 0.2) * H, math.sin(spireAngle) * curR))).Position
        end

        local diff = targetPos - part.Position
        local lockStrength = math.clamp(ringConfig.attractionStrength / 25, 15, 65)
        local vel = (diff * lockStrength) + Vector3.new(0, 14.46262424, 0)
        part.Velocity = vel
        part.RotVelocity = Vector3.zero
        pcall(function() part.AssemblyLinearVelocity = vel; part.AssemblyAngularVelocity = Vector3.zero; part.CanCollide = false end)
    end
end)

-- ========== RTX MODE (GRAFIK HD) ==========
do
    local R = { on = false, k = 0.5, saved = nil, fx = {}, q = nil, tech = nil }
    local PROPS = {
        "Brightness", "Ambient", "OutdoorAmbient", "ExposureCompensation",
        "EnvironmentDiffuseScale", "EnvironmentSpecularScale", "GlobalShadows", "ShadowSoftness",
    }

    local function fx(class, name)
        local o = R.fx[name]
        if not o or not o.Parent then
            o = Instance.new(class)
            o.Name = "VK_RTX_" .. name
            o.Parent = Lighting
            R.fx[name] = o
        end
        return o
    end

    local function applyRTX()
        local k = R.k
        Lighting.Brightness = 2 + 1.2 * k
        Lighting.ExposureCompensation = 0.15 + 0.25 * k
        Lighting.EnvironmentDiffuseScale = 0.5 + 0.5 * k
        Lighting.EnvironmentSpecularScale = 0.5 + 0.5 * k
        Lighting.GlobalShadows = true
        Lighting.OutdoorAmbient = Color3.fromRGB(90, 90, 100)
        pcall(function() Lighting.ShadowSoftness = 0.2 end)

        local b = fx("BloomEffect", "Bloom")
        b.Intensity = 0.25 + 0.5 * k
        b.Size = 28
        b.Threshold = 0.85

        local c = fx("ColorCorrectionEffect", "CC")
        c.Saturation = 0.1 + 0.25 * k
        c.Contrast = 0.08 + 0.15 * k
        c.TintColor = Color3.fromRGB(255, 248, 240)

        local s = fx("SunRaysEffect", "Rays")
        s.Intensity = 0.05 + 0.15 * k
        s.Spread = 0.7

        local d = fx("DepthOfFieldEffect", "DOF")
        d.FarIntensity = 0.1 + 0.15 * k
        d.FocusDistance = 80
        d.InFocusRadius = 60
        d.NearIntensity = 0

        local atm = Lighting:FindFirstChildOfClass("Atmosphere")
        if not atm or atm.Name:find("^VK_RTX_") then
            local a = fx("Atmosphere", "Atmo")
            a.Density = 0.15 + 0.1 * k
            a.Offset = 0.2
            a.Haze = 1 * k
        end
    end

    addHeader(tabMisc, "GRAFIK (JANGAN AKTIFKAN BERSAMAAN DENGAN BOOST FPS)")
    tabMisc:AddToggle("RTX Mode (Grafik HD)", false, function(on)
        R.on = on
        if on then
            R.saved = {}
            for _, p in ipairs(PROPS) do
                pcall(function() R.saved[p] = Lighting[p] end)
            end
            pcall(function() R.tech = gethiddenproperty(Lighting, "Technology") end)
            pcall(function() sethiddenproperty(Lighting, "Technology", Enum.Technology.Future) end)
            pcall(function()
                R.q = settings().Rendering.QualityLevel
                settings().Rendering.QualityLevel = Enum.QualityLevel.Level21
            end)
            applyRTX()
        else
            if R.saved then
                for p, v in pairs(R.saved) do pcall(function() Lighting[p] = v end) end
            end
            if R.tech then pcall(function() sethiddenproperty(Lighting, "Technology", R.tech) end) end
            if R.q then pcall(function() settings().Rendering.QualityLevel = R.q end) end

-- ========== BOOST FPS ==========
do
    local B = { on = false, orig = setmetatable({}, {__mode = "k"}), conn = nil, q = nil, shadows = true, terr = nil, token = 0 }
    local U = { on = false, orig = setmetatable({}, {__mode = "k"}), conn = nil, token = 0 }
    local EFFECT = { ParticleEmitter = true, Trail = true, Beam = true, Smoke = true, Fire = true, Sparkles = true }
    local POST = { BlurEffect = true, BloomEffect = true, SunRaysEffect = true, DepthOfFieldEffect = true, ColorCorrectionEffect = true }

    local function sweep(roots, fn, token, getToken)
        local n = 0
        for _, root in ipairs(roots) do
            for _, v in ipairs(root:GetDescendants()) do
                if getToken() ~= token then return end
                fn(v)
                n += 1
                if n % 400 == 0 then task.wait() end
            end
        end
    end

    local function boostObj(v)
        if v:IsA("BasePart") and not v:IsA("Terrain") then
            if not B.orig[v] then B.orig[v] = {v.Material, v.CastShadow, v.Reflectance} end
            v.Material = Enum.Material.SmoothPlastic
            v.CastShadow = false
            v.Reflectance = 0
        end
    end

    local function ultraObj(v)
        local cls = v.ClassName
        if EFFECT[cls] or (POST[cls] and not v.Name:find("^VK_")) then
            if U.orig[v] == nil then U.orig[v] = v.Enabled end
            v.Enabled = false
        elseif cls == "Decal" or cls == "Texture" then
            if v.Parent and v.Parent.Name == "Head" then return end
            if U.orig[v] == nil then U.orig[v] = v.Transparency end
            v.Transparency = 1
        end
    end

    addHeader(tabMisc, "BOOST FPS")
    tabMisc:AddToggle("Boost FPS (Material & Bayangan)", false, function(on)
        B.on = on
        B.token += 1
        if on then
            pcall(function()
                B.q = settings().Rendering.QualityLevel
                settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
            end)
            B.shadows = Lighting.GlobalShadows
            Lighting.GlobalShadows = false
            pcall(function()
                local t = workspace.Terrain
                B.terr = {t.WaterWaveSize, t.WaterWaveSpeed, t.WaterReflectance, t.Decoration}
                t.WaterWaveSize = 0
                t.WaterWaveSpeed = 0
                t.WaterReflectance = 0
                t.Decoration = false
            end)
            B.conn = workspace.DescendantAdded:Connect(function(v) if B.on then boostObj(v) end end)
            local tok = B.token
            task.spawn(function()
                sweep({workspace}, boostObj, tok, function() return B.token end)
            end)
        else
            if B.conn then B.conn:Disconnect(); B.conn = nil end
            for part, o in pairs(B.orig) do
                if part.Parent then
                    part.Material, part.CastShadow, part.Reflectance = o[1], o[2], o[3]
                end
            end
            B.orig = setmetatable({}, {__mode = "k"})
            if B.q then pcall(function() settings().Rendering.QualityLevel = B.q end) end
            Lighting.GlobalShadows = B.shadows
            if B.terr then
                pcall(function()
                    local t = workspace.Terrain
                    t.WaterWaveSize, t.WaterWaveSpeed, t.WaterReflectance, t.Decoration = B.terr[1], B.terr[2], B.terr[3], B.terr[4]
                end)
            end
        end
    end)

    tabMisc:AddToggle("Boost FPS Ultra (Hapus Efek & Tekstur)", false, function(on)
        U.on = on
        U.token += 1
        if on then
            U.conn = game.DescendantAdded:Connect(function(v) if U.on then ultraObj(v) end end)
            local tok = U.token
            task.spawn(function()
                sweep({workspace, Lighting}, ultraObj, tok, function() return U.token end)
            end)
        else
            if U.conn then U.conn:Disconnect(); U.conn = nil end
            for v, val in pairs(U.orig) do
                if v.Parent then
                    local cls = v.ClassName
                    if EFFECT[cls] or POST[cls] then v.Enabled = val else v.Transparency = val end
                end
            end
            U.orig = setmetatable({}, {__mode = "k"})
        end
    end)

    local gfxFirst = true
    local capFn = setfpscap or set_fps_cap
    tabMisc:AddStepper("FPS Cap (30-240)", 30, 240, 30, 60, function(v)
        if gfxFirst then gfxFirst = false; return end
        if capFn then pcall(capFn, v) end
    end)
    tabMisc:AddButton("Buka Batas FPS (Unlimited)", function()
        if capFn then pcall(capFn, 999) end
    end)
    tabMisc:AddToggle("Layar Hemat Baterai (Matikan Render 3D)", false, function(on)
        pcall(function() RunService:Set3dRenderingEnabled(not on) end)
    end)
end
