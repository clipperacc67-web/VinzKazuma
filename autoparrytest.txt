-- MAID PATTERN
-- ============================================================
local Maid = {}
Maid.__index = Maid
function Maid.new()
    return setmetatable({ _tasks = {} }, Maid)
end
function Maid:Give(task)
    if task ~= nil then table.insert(self._tasks, task) end
    return task
end
function Maid:Clean()
    for i = #self._tasks, 1, -1 do
        local t = self._tasks[i]
        pcall(function()
            if typeof(t) == "RBXScriptConnection" then
                t:Disconnect()
            elseif typeof(t) == "Instance" then
                t:Destroy()
            elseif type(t) == "function" then
                t()
            elseif type(t) == "table" then
                if t.Clean then t:Clean()
                elseif t.Destroy then t:Destroy()
                elseif t.Disconnect then t:Disconnect()
                end
            end
        end)
        self._tasks[i] = nil
    end
end

    if gethui then
        local ok, hui = pcall(gethui)

local function corner(parent, r)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, r or 6)
    c.Parent = parent
    return c
end

local function stroke(parent, color, thick, trans)
    local s = Instance.new("UIStroke")
    s.Color           = color or Theme.Border
    s.Thickness       = thick or 1
    s.Transparency    = trans or 0.35
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    s.Parent          = parent
    return s
end

local function makeDraggable(frame, handle, maid)
    maid = maid or Maid.new()
    handle = handle or frame
    local dragging, dragStart, startPos

    maid:Give(handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging   = true
            dragStart  = input.Position
            startPos   = frame.Position
        end
    end))

    maid:Give(UserInputService.InputChanged:Connect(function(input)
        if not dragging then return end
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end))

    maid:Give(UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end))

    return maid
end

-- ---------- Toast ----------
local function repositionToasts()
    for i, t in ipairs(Lib._activeToasts) do
        if t and t.Parent then
            TweenService:Create(t, Theme.Tween, {
                Position = UDim2.new(1, -300, 1, -80 - (i - 1) * 60),
            }):Play()
        end
    end
end

function Lib.toast(msg, kind)
    kind = kind or "info"
    local color = Theme.Accent
    if kind == "success" then color = Theme.Success
    elseif kind == "error" then color = Theme.Danger
    elseif kind == "warning" then color = Theme.Warning
    end

    if not Lib._toastGui or not Lib._toastGui.Parent then
        local sg = Instance.new("ScreenGui")
        sg.Name           = "AutoCombat_Toast"
        sg.ResetOnSpawn   = false
        sg.IgnoreGuiInset = true
        sg.DisplayOrder   = 999
        sg.Parent         = getGuiParent()
        Lib._toastGui = sg
        GlobalMaid:Give(sg)
    end

    local toast = Instance.new("Frame")
    toast.Size                   = UDim2.fromOffset(280, 52)
    toast.Position               = UDim2.new(1, 20, 1, -80)
    toast.BackgroundColor3       = Theme.Surface
    toast.BackgroundTransparency = 0.08
    toast.BorderSizePixel        = 0
    toast.Parent                 = Lib._toastGui
    corner(toast, 8)
    stroke(toast, color, 1, 0.3)

    local accent = Instance.new("Frame")
    accent.Size             = UDim2.fromOffset(4, 52)
    accent.BackgroundColor3 = color
    accent.BorderSizePixel  = 0
    accent.Parent           = toast
    corner(accent, 8)

    local tl = Instance.new("TextLabel")
    tl.Size                   = UDim2.new(1, -24, 1, 0)
    tl.Position               = UDim2.fromOffset(16, 0)
    tl.BackgroundTransparency = 1
    tl.Text                   = msg
    tl.TextColor3             = Theme.Text
    tl.Font                   = Theme.Font
    tl.TextSize               = 12
    tl.TextXAlignment         = Enum.TextXAlignment.Left
    tl.TextWrapped            = true
    tl.Parent                 = toast

    table.insert(Lib._activeToasts, toast)
    repositionToasts()

    task.delay(3, function()
        if not toast or not toast.Parent then return end
        for i, t in ipairs(Lib._activeToasts) do
            if t == toast then
                table.remove(Lib._activeToasts, i)
                break
            end
        end
        TweenService:Create(toast, Theme.Tween, {
            Position             = UDim2.new(1, 20, 1, toast.Position.Y.Offset),
            BackgroundTransparency = 1,
        }):Play()
        task.delay(0.3, function()
            if toast then toast:Destroy() end
            repositionToasts()
        end)
    end)

    return toast
end

function Lib.addToggle(parent, opts)
    opts = opts or {}
    local state = { value = opts.default or false }

    local frame = Instance.new("Frame")
    frame.Size                   = UDim2.new(1, 0, 0, 36)
    frame.BackgroundColor3       = Theme.Surface
    frame.BackgroundTransparency = 0.4
    frame.BorderSizePixel        = 0
    frame.Parent                 = parent
    corner(frame, 8)
    stroke(frame, Theme.Border, 1, 0.6)

    local statusDot = Instance.new("Frame")
    statusDot.Size             = UDim2.fromOffset(6, 6)
    statusDot.Position         = UDim2.new(0, 8, 0.5, -3)
    statusDot.BackgroundColor3 = Theme.TextDim
    statusDot.BorderSizePixel  = 0
    statusDot.Parent           = frame
    corner(statusDot, 3)

    local label = Instance.new("TextLabel")
    label.Size                   = UDim2.new(1, -78, 1, 0)
    label.Position               = UDim2.fromOffset(22, 0)
    label.BackgroundTransparency = 1
    label.Text                   = opts.text or "Toggle"
    label.TextColor3             = Theme.Text
    label.Font                   = Theme.Font
    label.TextSize               = 13
    label.TextXAlignment         = Enum.TextXAlignment.Left
    label.Parent                 = frame

    local track = Instance.new("Frame")
    track.Size             = UDim2.fromOffset(42, 20)
    track.Position         = UDim2.new(1, -54, 0.5, -10)
    track.BackgroundColor3 = Theme.SurfaceAlt
    track.BorderSizePixel  = 0
    track.Parent           = frame
    corner(track, 10)

    local knob = Instance.new("Frame")
    knob.Size             = UDim2.fromOffset(14, 14)
    knob.Position         = UDim2.fromOffset(3, 3)
    knob.BackgroundColor3 = Theme.Text
    knob.BorderSizePixel  = 0
    knob.Parent           = track
    corner(knob, 7)

    local function apply(v, animate)
        state.value = v
        local knobPos   = v and UDim2.fromOffset(25, 3) or UDim2.fromOffset(3, 3)
        local trackCol  = v and Theme.Accent or Theme.SurfaceAlt
        local dotCol    = v and Theme.Success or Theme.TextDim
        if animate then
            TweenService:Create(knob,      Theme.Tween, { Position = knobPos }):Play()
            TweenService:Create(track,     Theme.Tween, { BackgroundColor3 = trackCol }):Play()
            TweenService:Create(statusDot, Theme.Tween, { BackgroundColor3 = dotCol }):Play()
        else
            knob.Position           = knobPos
            track.BackgroundColor3  = trackCol
            statusDot.BackgroundColor3 = dotCol
        end
        if opts.callback then task.spawn(function() opts.callback(v) end) end
    end

    local hit = Instance.new("TextButton")
    hit.Size                   = UDim2.new(1, 0, 1, 0)
    hit.BackgroundTransparency = 1
    hit.Text                   = ""
    hit.ZIndex                 = 2
    hit.Parent                 = frame
    hit.MouseButton1Click:Connect(function() apply(not state.value, true) end)

    apply(state.value, false)
    return {
        set = function(_, v) apply(v, true) end,
        get = function(_) return state.value end,
    }
end

-- ---------- Slider ----------
function Lib.addSlider(parent, opts)
    opts = opts or {}
    local minVal = opts.min or 0
    local maxVal = opts.max or 100
    local step   = opts.step or 0
    local current = math.clamp(opts.default or minVal, minVal, maxVal)

    local frame = Instance.new("Frame")
    frame.Size                   = UDim2.new(1, 0, 0, 54)
    frame.BackgroundColor3       = Theme.Surface
    frame.BackgroundTransparency = 0.4
    frame.BorderSizePixel        = 0
    frame.Parent                 = parent
    corner(frame, 8)
    stroke(frame, Theme.Border, 1, 0.6)

    local label = Instance.new("TextLabel")
    label.Size                   = UDim2.new(1, -90, 0, 20)
    label.Position               = UDim2.fromOffset(12, 6)
    label.BackgroundTransparency = 1
    label.Text                   = opts.text or "Slider"
    label.TextColor3             = Theme.Text
    label.Font                   = Theme.Font
    label.TextSize               = 13
    label.TextXAlignment         = Enum.TextXAlignment.Left
    label.Parent                 = frame

    local valueLabel = Instance.new("TextLabel")
    valueLabel.Size                   = UDim2.fromOffset(72, 20)
    valueLabel.Position               = UDim2.new(1, -84, 0, 6)
    valueLabel.BackgroundTransparency = 1
    valueLabel.Text                   = tostring(current)
    valueLabel.TextColor3             = Theme.Accent
    valueLabel.Font                   = Theme.FontBold
    valueLabel.TextSize               = 13
    valueLabel.TextXAlignment         = Enum.TextXAlignment.Right
    valueLabel.Parent                 = frame

    local track = Instance.new("Frame")
    track.Size             = UDim2.new(1, -24, 0, 6)
    track.Position         = UDim2.new(0, 12, 1, -18)
    track.BackgroundColor3 = Theme.SurfaceAlt
    track.BorderSizePixel  = 0
    track.Parent           = frame
    corner(track, 3)

    local fill = Instance.new("Frame")
    fill.Size             = UDim2.new(0, 0, 1, 0)
    fill.BackgroundColor3 = Theme.Accent
    fill.BorderSizePixel  = 0
    fill.Parent           = track
    corner(fill, 3)

    local knob = Instance.new("Frame")
    knob.Size             = UDim2.fromOffset(14, 14)
    knob.AnchorPoint      = Vector2.new(0.5, 0.5)
    knob.Position         = UDim2.new(0, 0, 0.5, 0)
    knob.BackgroundColor3 = Theme.Text
    knob.BorderSizePixel  = 0
    knob.Parent           = track
    corner(knob, 7)
    stroke(knob, Theme.Accent, 2, 0)

    local function snap(v)
        if step > 0 then return math.floor(v / step + 0.5) * step end
        return v
    end

    local function update(v, animate, fire)
        v = math.clamp(snap(v), minVal, maxVal)
        current = v
        local ratio = (maxVal > minVal) and (v - minVal) / (maxVal - minVal) or 0
        local tSize = UDim2.new(ratio, 0, 1, 0)
        local tPos  = UDim2.new(ratio, 0, 0.5, 0)
        if animate then
            TweenService:Create(fill, Theme.TweenFast, { Size = tSize }):Play()
            TweenService:Create(knob, Theme.TweenFast, { Position = tPos }):Play()
        else
            fill.Size   = tSize
            knob.Position = tPos
        end
        valueLabel.Text = tostring(math.floor(v * 100 + 0.5) / 100)
        if fire and opts.callback then task.spawn(function() opts.callback(v) end) end
    end

    local dragging = false
    local function setFromMouse(xPos)
        local ax = track.AbsolutePosition.X
        local w  = track.AbsoluteSize.X
        if w <= 0 then return end
        local ratio = math.clamp((xPos - ax) / w, 0, 1)
        update(minVal + ratio * (maxVal - minVal), false, true)
    end

    local hit = Instance.new("TextButton")
    hit.Size                   = UDim2.new(1, 0, 1, 0)
    hit.BackgroundTransparency = 1
    hit.Text                   = ""
    hit.ZIndex                 = 2
    hit.Parent                 = frame

    hit.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            setFromMouse(input.Position.X)
        end
    end)

    GlobalMaid:Give(UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch) then
            setFromMouse(input.Position.X)
        end
    end))

    GlobalMaid:Give(UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end))

    update(current, false, false)
    return {
        set = function(_, v) update(v, true, true) end,
        get = function(_) return current end,
    }
end

    local arrow = Instance.new("TextLabel")
    arrow.Size                   = UDim2.fromOffset(16, 16)
    arrow.Position               = UDim2.new(1, -22, 0.5, -8)
    arrow.BackgroundTransparency = 1
    arrow.Text                   = "▾"
    arrow.TextColor3             = Theme.TextDim
    arrow.TextSize               = 12
    arrow.Parent                 = header

    local list = Instance.new("Frame")
    list.Size                   = UDim2.new(1, 0, 0, 0)
    list.Position               = UDim2.new(0, 0, 1, 4)
    list.BackgroundColor3       = Theme.Surface
    list.BackgroundTransparency = 0.05
    list.BorderSizePixel        = 0
    list.ClipsDescendants       = true
    list.Visible                = false
    list.ZIndex                 = 20
    list.Parent                 = container
    corner(list, 8)
    stroke(list, Theme.Accent, 1, 0.4)

    local listLayout = Instance.new("UIListLayout")
    listLayout.Padding   = UDim.new(0, 2)
    listLayout.SortOrder = Enum.SortOrder.LayoutOrder
    listLayout.Parent    = list

    local listPad = Instance.new("UIPadding")
    listPad.PaddingTop    = UDim.new(0, 4)
    listPad.PaddingBottom = UDim.new(0, 4)
    listPad.Parent        = list

    local optionButtons = {}

    local function setValue(v, fire)
        current = v
        currentLabel.Text = tostring(v)
        for opt, btn in pairs(optionButtons) do
            if opt == v then
                btn.BackgroundColor3 = Theme.Accent
                btn.TextColor3       = Theme.Text
            else
                btn.BackgroundColor3 = Theme.SurfaceAlt
                btn.TextColor3       = Theme.TextDim
            end
        end
        if fire and opts.callback then task.spawn(function() opts.callback(v) end) end
    end

    local function close()
        expanded = false
        TweenService:Create(list, Theme.Tween, { Size = UDim2.new(1, 0, 0, 0) }):Play()
        TweenService:Create(arrow, Theme.Tween, { Rotation = 0 }):Play()
        task.delay(0.25, function() if list then list.Visible = false end end)
    end

    for _, opt in ipairs(options) do
        local ob = Instance.new("TextButton")
        ob.Size             = UDim2.new(1, -8, 0, 26)
        ob.Position         = UDim2.fromOffset(4, 0)
        ob.BackgroundColor3 = Theme.SurfaceAlt

        ob.Text             = "  " .. tostring(opt)
        ob.TextColor3       = Theme.TextDim
        ob.Font             = Theme.Font
        ob.TextSize         = 12
        ob.TextXAlignment   = Enum.TextXAlignment.Left
        ob.AutoButtonColor  = false
        ob.ZIndex           = 21
        ob.Parent           = list
        corner(ob, 6)
        optionButtons[opt] = ob

        ob.MouseEnter:Connect(function()
            if current ~= opt then
                TweenService:Create(ob, Theme.TweenFast, { BackgroundColor3 = Theme.Border }):Play()
            end
        end)
        ob.MouseLeave:Connect(function()
            if current ~= opt then
                TweenService:Create(ob, Theme.TweenFast, { BackgroundColor3 = Theme.SurfaceAlt }):Play()
            end
        end)
        ob.MouseButton1Click:Connect(function()
            setValue(opt, true)
            close()
        end)
    end

    local totalH = 4 + #options * 28 + 4
    setValue(current, false)

    header.MouseButton1Click:Connect(function()
        expanded = not expanded
        if expanded then
            list.Visible = true
            TweenService:Create(list, Theme.Tween, { Size = UDim2.new(1, 0, 0, totalH) }):Play()
            TweenService:Create(arrow, Theme.Tween, { Rotation = 180 }):Play()
        else
            close()
        end
    end)

    return {
        set = function(_, v) setValue(v, true) end,
        get = function(_) return current end,
    }
end

local State = {
    AutoParry     = false,
    ParryMode     = "Directional",
    ParryRange    = 12,
    ParryKey      = Enum.KeyCode.F,
    ParryCooldown = 0.3,
    LastParryTime = 0,

    AutoDodge     = false,
    DodgeMethod   = "Dash",
    DodgeRange    = 15,
    DodgeDelay    = 0.10,
    DodgeDistance = 12,
    DodgeCooldown = 0.50,
    LastDodgeTime = 0,

    PingCompensation  = 1.0,
    VisualizerEnabled = true,

    FPS         = 60,
    ThreatCount = 0,
}

-- ============================================================
-- UTIL: PING & ROOT
-- ============================================================
local function getPing()
    local ok, ping = pcall(function()
        return Stats.Network.ServerStatsItem["Data Ping"]:GetValue() / 1000
    end)
    return ok and ping or 0.05
end

local function getRoot()
    local char = LocalPlayer.Character
    if not char then return nil, nil end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local root = char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso")
    return root, char, hum
end

-- ============================================================
-- THREAT SCANNER
-- ============================================================
local function findThreats(range)
    local root, char = getRoot()
    if not root then return {} end
    local myPos = root.Position

    local threats = {}
    for _, part in ipairs(Workspace:GetDescendants()) do
        if part:IsA("BasePart") and not part.Anchored and part ~= root then
            if not (char and part:IsDescendantOf(char)) then
                local vel = part.AssemblyLinearVelocity or part.Velocity
                if vel.Magnitude > 10 then
                    local toMe = myPos - part.Position
                    local dist = toMe.Magnitude
                    if dist > 0.5 and dist <= range then
                        local dirToMe = toMe.Unit
                        local velDir  = vel.Unit
                        local dot     = dirToMe:Dot(velDir)
                        if dot > 0.5 then
                            local closing = vel.Magnitude * dot
                            table.insert(threats, {
                                part         = part,
                                distance     = dist,
                                velocity     = vel.Magnitude,
                                closingSpeed = closing,
                                dirToMe      = dirToMe,
                                dot          = dot,
                                eta          = dist / math.max(closing, 1),
                            })
                        end
                    end
                end
            end
        end
    end
    table.sort(threats, function(a, b) return a.eta < b.eta end)
    return threats
end

-- ============================================================
-- KINEMATIC SOLVER
-- ============================================================
local function dynamicThreshold(baseDist, velocity)
    local ping = getPing()
    return baseDist + (velocity * ping * State.PingCompensation)
end

local function computeDodgeVector(threat)
    local dir = threat.dirToMe
    local up  = Vector3.new(0, 1, 0)
    local lateral = dir:Cross(up)
    if lateral.Magnitude < 0.05 then
        lateral = dir:Cross(Vector3.new(1, 0, 0))
    end
    lateral = lateral.Unit

    local root = getRoot()
    if root then
        local v = root.AssemblyLinearVelocity or root.Velocity
        if v.Magnitude > 1 and v:Dot(lateral) < 0 then
            lateral = -lateral
        elseif math.random() > 0.5 then
            lateral = -lateral
        end
    end
    return lateral
end

-- ============================================================
-- ACTUATORS
-- ============================================================
local function triggerParry()
    local now = tick()
    if now - State.LastParryTime < State.ParryCooldown then return end
    State.LastParryTime = now
    pcall(function()
        VirtualInputManager:SendKeyEvent(true,  State.ParryKey, false, game)
        task.wait(0.03)
        VirtualInputManager:SendKeyEvent(false, State.ParryKey, false, game)
    end)
end

local function performDodge(method, lateralDir, distance)
    local root = getRoot()
    if not root then return end

    if method == "Teleport" then
        local newCF = CFrame.new(root.Position + lateralDir * distance) * (root.CFrame - root.CFrame.Position)
        root.CFrame = newCF
    elseif method == "Dash" then
        local bv = Instance.new("BodyVelocity")
        bv.Velocity = lateralDir * 90
        bv.MaxForce = Vector3.new(1e5, 0, 1e5)
        bv.Parent   = root
        task.delay(0.15, function() if bv and bv.Parent then bv:Destroy() end end)
    elseif method == "Velocity" then
        local bv = Instance.new("BodyVelocity")
        bv.Velocity = lateralDir * 60
        bv.MaxForce = Vector3.new(math.huge, 0, math.huge)
        bv.Parent   = root
        task.delay(0.25, function() if bv and bv.Parent then bv:Destroy() end end)
    end
end

local function triggerDodge(threat)
    local now = tick()
    if now - State.LastDodgeTime < State.DodgeCooldown then return end
    State.LastDodgeTime = now
    local lateral = computeDodgeVector(threat)
    performDodge(State.DodgeMethod, lateral, State.DodgeDistance)
end

local function shouldParry(threat)
    if State.ParryMode == "Instant" then return true end
    if State.ParryMode == "Directional" then
        local root = getRoot()
        if not root then return false end
        local look     = root.CFrame.LookVector
        local toThreat = (threat.part.Position - root.Position)
        if toThreat.Magnitude < 0.1 then return false end
        return look:Dot(toThreat.Unit) > 0.3
    end
    if State.ParryMode == "Legit" then return true end
    return false
end

-- ============================================================
-- CORE HEARTBEAT
-- ============================================================
local function onHeartbeat(dt)
    State.FPS = State.FPS * 0.9 + (1 / math.max(dt, 0.001)) * 0.1

    if not State.AutoParry and not State.AutoDodge then
        State.ThreatCount = 0
        return
    end

    local root = getRoot()
    if not root then
        State.ThreatCount = 0
        return
    end

    local scanRange = math.max(
        State.AutoParry and State.ParryRange or 0,
        State.AutoDodge and State.DodgeRange or 0
    )
    if scanRange <= 0 then return end

    local threats = findThreats(scanRange)
    State.ThreatCount = #threats

    local first = threats[1]
    if not first then return end

    if State.AutoParry then
        local threshold = dynamicThreshold(State.ParryRange, first.velocity)
        if first.distance <= threshold and shouldParry(first) then
            if State.ParryMode == "Legit" then
                local delay = math.random() * 0.08 + 0.02
                task.delay(delay, function()
                    if State.AutoParry then triggerParry() end
                end)
            else
                triggerParry()
            end
        end
    end

    if State.AutoDodge then
        local threshold = dynamicThreshold(State.DodgeRange, first.velocity)
        if first.distance <= threshold then
            task.delay(State.DodgeDelay, function()
                if State.AutoDodge then triggerDodge(first) end
            end)
        end
    end
end

local Visualizer = { _part = nil }

function Visualizer:Create()
    self:Destroy()
    local root = getRoot()
    if not root then return end

    local part = Instance.new("Part")
    part.Name         = "AutoCombat_Visualizer"
    part.Shape        = Enum.PartType.Cylinder
    part.Size         = Vector3.new(0.15, State.ParryRange * 2, State.ParryRange * 2)
    part.Material     = Enum.Material.ForceField
    part.Color        = Theme.Accent
    part.Transparency = 0.6
    part.CanCollide   = false
    part.CanQuery     = false
    part.CanTouch     = false
    part.Anchored     = true
    part.Parent       = Workspace
    self._part = part
    GlobalMaid:Give(part)
end

function Visualizer:Update()
    if not State.VisualizerEnabled then
        if self._part then self:Destroy() end
        return
    end
    if not self._part or not self._part.Parent then
        self:Create()
        return
    end
    local root = getRoot()
    if not root then return end

    local pos = root.Position
    self._part.CFrame = CFrame.new(pos) * CFrame.Angles(0, 0, math.rad(90))
    self._part.Size   = Vector3.new(0.15, State.ParryRange * 2, State.ParryRange * 2)

    local hasThreat   = (State.ThreatCount or 0) > 0
    local targetColor = hasThreat and Theme.Danger or Theme.Accent
    local targetTrans = hasThreat and 0.3 or 0.6
    self._part.Color        = self._part.Color:Lerp(targetColor, 0.2)
    self._part.Transparency = self._part.Transparency + (targetTrans - self._part.Transparency) * 0.2
end

function Visualizer:Destroy()
    if self._part then
        self._part:Destroy()
        self._part = nil
    end
end
