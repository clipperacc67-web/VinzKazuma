local function Corner(parent, radius)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, radius or 8)
    c.Parent = parent
    return c
end
local function Stroke(parent, color, thickness, transparency)
    local s = Instance.new("UIStroke")
    s.Color = color or Theme.Border
    s.Thickness = thickness or 1
    s.Transparency = transparency or 0
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    s.Parent = parent
    return s
end
local function Gradient(parent, c1, c2, rot)
    local g = Instance.new("UIGradient")
    g.Color = ColorSequence.new(c1 or Theme.Accent, c2 or Theme.Accent2)
    g.Rotation = rot or 90
    g.Parent = parent
    return g
end
local function Tween(obj, props, dur, style, dir)
    local t = TweenService:Create(obj,
        TweenInfo.new(dur or 0.2, style or Enum.EasingStyle.Quad, dir or Enum.EasingDirection.Out),
        props)
    t:Play()
    return t
end

    btn.TextColor3 = Theme.Muted
    btn.AutoButtonColor = false
    btn.LayoutOrder = order
    btn.Parent = tabBar
    Corner(btn, 8)

    local page = Instance.new("ScrollingFrame")
    page.Size = UDim2.fromScale(1, 1)
    page.BackgroundTransparency = 1
    page.BorderSizePixel = 0
    page.ScrollBarThickness = 3
    page.ScrollBarImageColor3 = Theme.Accent
    page.CanvasSize = UDim2.new(0, 0, 0, 0)
    page.AutomaticCanvasSize = Enum.AutomaticSize.Y
    page.Visible = false
    page.Parent = pagesHolder

    local pageLayout = Instance.new("UIListLayout")
    pageLayout.Padding = UDim.new(0, 8)
    pageLayout.SortOrder = Enum.SortOrder.LayoutOrder
    pageLayout.Parent = page

    local function setActive()
        if currentTab then
            currentTab.btn.BackgroundColor3 = Theme.Panel
            currentTab.btn.TextColor3 = Theme.Muted
            currentTab.page.Visible = false
        end
        btn.BackgroundColor3 = Theme.Accent
        btn.TextColor3 = Theme.Text
        page.Visible = true
        currentTab = {btn = btn, page = page}
    end

    btn.MouseButton1Click:Connect(setActive)
    tabs[name] = {btn = btn, page = page, setActive = setActive, layout = pageLayout}
    return tabs[name]
end

    valLbl.Position = UDim2.new(1, -112, 0, 6)

    valLbl.Font = Theme.FontBold

    valLbl.TextXAlignment = Enum.TextXAlignment.Right
    valLbl.Parent = row

    local track = Instance.new("Frame")
    track.Size = UDim2.new(1, -24, 0, 6)
    track.Position = UDim2.new(0, 12, 0, 36)
    track.BackgroundColor3 = Theme.PanelAlt
    track.BorderSizePixel = 0
    track.Parent = row
    Corner(track, 3)

    local fill = Instance.new("Frame")
    fill.Size = UDim2.new((default - min) / (max - min), 0, 1, 0)
    fill.BackgroundColor3 = Theme.Accent
    fill.BorderSizePixel = 0
    fill.Parent = track
    Corner(fill, 3)
    Gradient(fill, Theme.Accent, Theme.Accent2, 0)

    local knob = Instance.new("Frame")
    knob.Size = UDim2.fromOffset(14, 14)
    knob.Position = UDim2.new(fill.Size.X.Scale, -7, 0.5, -7)
    knob.BackgroundColor3 = Theme.Text
    knob.BorderSizePixel = 0
    knob.Parent = track
    Corner(knob, 7)
    Stroke(knob, Theme.Accent, 2, 0)

    local value = default
    local dragging = false
    local function setFromX(x)
        local rel = math.clamp((x - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        value = min + (max - min) * rel
        if max - min <= 2 then value = math.floor(value * 100 + 0.5) / 100
        else value = math.floor(value + 0.5) end
        fill.Size = UDim2.new(rel, 0, 1, 0)
        knob.Position = UDim2.new(rel, -7, 0.5, -7)
        valLbl.Text = tostring(value) .. (suffix or "")
        if callback then task.spawn(callback, value) end
    end

    track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            setFromX(input.Position.X)
        end
    end)
    UIS.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            setFromX(input.Position.X)
        end
    end)
    UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    return {row = row, get = function() return value end, set = function(v)
        value = v
        local rel = (v - min) / (max - min)
        fill.Size = UDim2.new(rel, 0, 1, 0)
        knob.Position = UDim2.new(rel, -7, 0.5, -7)
        valLbl.Text = tostring(v) .. (suffix or "")
    end}
end

local Combat = { parryRemotes = {}, dodgeRemotes = {}, watchedCount = 0, hookConns = {} }

local PARRY_KW  = {"parry","block","perfect","counter","guard"}
local DODGE_KW  = {"dodge","roll","dash","evade","sidestep","iframe"}
local ATTACK_KW = {"attack","hit","swing","punch","strike","melee","combat","slash","stab","kick","damage","shot","shoot","fire","sword","weapon"}
local ANIM_KW   = {"punch","hit","swing","attack","kick","slash","stab","sword","combat","melee","strike","shot"}

local function NameHasAny(name, list)
    local n = string.lower(name)
    for _, k in ipairs(list) do if string.find(n, k, 1, true) then return true end end
    return false
end

local function CollectRemotes()
    Combat.parryRemotes = {}
    Combat.dodgeRemotes = {}
    Combat.watchedCount = 0
    local stack = {Replicated}
    local seen = {}
    while #stack > 0 do
        local node = table.remove(stack)
        if node and not seen[node] then
            seen[node] = true
            for _, child in ipairs(node:GetChildren()) do
                if child:IsA("RemoteEvent") or child:IsA("RemoteFunction") then
                    if NameHasAny(child.Name, PARRY_KW) then table.insert(Combat.parryRemotes, child) end
                    if NameHasAny(child.Name, DODGE_KW) then table.insert(Combat.dodgeRemotes, child) end
                    if NameHasAny(child.Name, ATTACK_KW) then Combat.watchedCount = Combat.watchedCount + 1 end
                elseif child:IsA("Folder") or child:IsA("Model") or child:IsA("Configuration") then
                    table.insert(stack, child)
                end
            end
        end
    end
end

function Combat.Scan()
    CollectRemotes()
    local pg = LP:FindFirstChild("PlayerGui")
    if pg then
        for _, d in ipairs(pg:GetDescendants()) do
            if d:IsA("RemoteEvent") then
                if NameHasAny(d.Name, PARRY_KW) then table.insert(Combat.parryRemotes, d) end
                if NameHasAny(d.Name, DODGE_KW) then table.insert(Combat.dodgeRemotes, d) end
            end
        end
    end
    statusText.Text = string.format("REMS %d/%d", #Combat.parryRemotes, #Combat.dodgeRemotes)
end

function Combat.FireParry()
    if #Combat.parryRemotes == 0 then return false end
    local now = tick()
    if now - State.LastParry < Config.ParryCooldown then return false end
    State.LastParry = now
    for _, remote in ipairs(Combat.parryRemotes) do
        if remote:IsA("RemoteEvent") then
            if not pcall(function() remote:FireServer() end) then
                pcall(function() remote:FireServer("Parry") end)
            end
        elseif remote:IsA("RemoteFunction") then
            pcall(function() remote:InvokeServer() end)
        end
    end
    Config.Stats.Parries = Config.Stats.Parries + 1
    if Config.VisualFeedback then
        local flash = Instance.new("Frame")
        flash.Size = UDim2.fromScale(1, 1)
        flash.BackgroundColor3 = Theme.Accent
        flash.BackgroundTransparency = 0.85
        flash.BorderSizePixel = 0
        flash.ZIndex = 999
        flash.Parent = gui
        Tween(flash, {BackgroundTransparency = 1}, 0.25)
        task.delay(0.3, function() flash:Destroy() end)
    end
    return true
end

function Combat.FireDodge()
    if #Combat.dodgeRemotes == 0 then return false end
    local now = tick()
    if now - State.LastDodge < Config.DodgeCooldown then return false end
    State.LastDodge = now
    local dir = Config.DodgeDirection
    for _, remote in ipairs(Combat.dodgeRemotes) do
        if remote:IsA("RemoteEvent") then
            if dir == "Random" then
                local opts = {"Back","Left","Right","Forward"}
                local pick = opts[math.random(1, #opts)]
                if not pcall(function() remote:FireServer(pick) end) then
                    pcall(function() remote:FireServer() end)
                end
            else
                if not pcall(function() remote:FireServer(dir) end) then
                    pcall(function() remote:FireServer() end)
                end
            end
        elseif remote:IsA("RemoteFunction") then
            pcall(function() remote:InvokeServer(dir) end)
        end
    end
    Config.Stats.Dodges = Config.Stats.Dodges + 1
    return true
end

local function IsOpponent(char)
    if not char then return false end
    local plr = Players:GetPlayerFromCharacter(char)
    return plr ~= nil and plr ~= LP
end

local function GetNearestOpponent()
    local nearest, nearestDist = nil, math.huge
    local myChar = LP.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil, math.huge end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP and plr.Character then
            local root = plr.Character:FindFirstChild("HumanoidRootPart")
            local hum = plr.Character:FindFirstChildOfClass("Humanoid")
            if root and hum and hum.Health > 0 then
                local d = (root.Position - myRoot.Position).Magnitude
                if d < nearestDist then nearest, nearestDist = plr, d end
            end
        end
    end
    return nearest, nearestDist
end

local function TriggerCombat()
    Config.Stats.Detections = Config.Stats.Detections + 1
    task.spawn(function()
        if Config.AutoParry then Combat.FireParry() end
        if Config.AutoDodge then
            task.wait(Config.ParryDelay)
            Combat.FireDodge()
        end
    end)
end

local function HookAttackRemotes()
    for _, c in ipairs(Combat.hookConns) do pcall(function() c:Disconnect() end) end
    table.clear(Combat.hookConns)
    local stack = {Replicated}
    local seen = {}
    while #stack > 0 do
        local node = table.remove(stack)
        if node and not seen[node] then
            seen[node] = true
            for _, child in ipairs(node:GetChildren()) do
                if child:IsA("RemoteEvent") and NameHasAny(child.Name, ATTACK_KW) then
                    local conn = child.OnClientEvent:Connect(function(...)
                        local args = {...}
                        local source
                        for _, a in ipairs(args) do
                            if typeof(a) == "Instance" and a:IsA("Player") then source = a; break end
                            if typeof(a) == "Instance" and a:IsA("Model") then
                                local p = Players:GetPlayerFromCharacter(a)
                                if p then source = p; break end
                            end
                        end
                        if source == LP then return end
                        if source and source ~= LP then
                            local myChar = LP.Character
                            local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
                            local srcRoot = source.Character and source.Character:FindFirstChild("HumanoidRootPart")
                            if myRoot and srcRoot then
                                if (myRoot.Position - srcRoot.Position).Magnitude <= Config.ParryRange then
                                    TriggerCombat()
                                end
                            else
                                TriggerCombat()
                            end
                        else
                            local _, d = GetNearestOpponent()
                            if d <= Config.ParryRange then TriggerCombat() end
                        end
                    end)
                    table.insert(Combat.hookConns, conn)
                elseif child:IsA("Folder") or child:IsA("Model") or child:IsA("Configuration") then
                    table.insert(stack, child)
                end
            end
        end
    end
end

local function WatchCharacter(char)
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum then return end
    local animator = hum:FindFirstChildOfClass("Animator")
    if not animator then return end
    local conn = animator.AnimationPlayed:Connect(function(track)
        if not IsOpponent(char) then return end
        local id = track.Animation and track.Animation.AnimationId or ""
        local nm = track.Name or ""
        if NameHasAny(nm, ANIM_KW) or NameHasAny(id, ANIM_KW) then
            local myChar = LP.Character
            local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
            local tRoot = char:FindFirstChild("HumanoidRootPart")
            if myRoot and tRoot then
                if (myRoot.Position - tRoot.Position).Magnitude <= Config.ParryRange then
                    TriggerCombat()
                end
            end
        end
    end)
    table.insert(State.Connections, conn)
end

local function AttachCharacterListeners()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LP then
            if plr.Character then WatchCharacter(plr.Character) end
            track(plr.CharacterAdded:Connect(function(c) task.wait(0.5); WatchCharacter(c) end))
        end
    end
    track(Players.PlayerAdded:Connect(function(plr)
        plr.CharacterAdded:Connect(function(c) task.wait(0.5); WatchCharacter(c) end)
    end))
end

local function AttributeWatch()
    local char = LP.Character
    if not char then return end
    track(char.AttributeChanged:Connect(function(attr)
        local a = string.lower(attr)
        if a:find("parry") or a:find("perfect") or a:find("window") then
            local val = char:GetAttribute(attr)
            if val == true then TriggerCombat() end
        end
    end))
end

local function RefreshNearest()
    local n, d = GetNearestOpponent()
    State.Nearest = n
    State.NearestDist = d
end
