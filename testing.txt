function VK.bind(conn, key)
	table.insert(VK.Conn, {c = conn, k = key})
	return conn
end

function VK.unbind(key)
	for i = #VK.Conn, 1, -1 do
		if key == nil or VK.Conn[i].k == key then
			pcall(function() VK.Conn[i].c:Disconnect() end)
			table.remove(VK.Conn, i)
		end
	end
end

function VK.safe(fn)
	local ok, err = pcall(fn)
	if not ok then warn("[VinzKazuma] " .. tostring(err)) end
	return ok
end

function VK.hum()

function VK.isAlive(plr)
	local c = plr.Character
	local h = c and c:FindFirstChildOfClass("Humanoid")

function VK.new(class, props, parent)
	local inst = Instance.new(class)
	for k, v in pairs(props) do
		if k ~= "Parent" then inst[k] = v end
	end
	inst.Parent = parent
	return inst
end

function VK.corner(inst, r)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, r or 8)
	c.Parent = inst
	return c
end

function VK.stroke(inst, col, th)
	local s = Instance.new("UIStroke")
	s.Color = col or VK.Border
	s.Thickness = th or 1
	s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	s.Parent = inst
	return s
end

function VK.pad(inst, n)
	local p = Instance.new("UIPadding")
	p.PaddingTop = UDim.new(0, n)
	p.PaddingBottom = UDim.new(0, n)
	p.PaddingLeft = UDim.new(0, n)
	p.PaddingRight = UDim.new(0, n)
	p.Parent = inst
	return p
end

function VK.showPage(key)
	for _, item in pairs(navButtons) do
		local on = item.key == key
		item.btn.TextColor3 = on and VK.Text or VK.Dim
		item.btn.BackgroundTransparency = on and 0.85 or 1
		item.btn.BackgroundColor3 = on and VK.Card or VK.Panel
		item.marker.Visible = on
	end
	for k, p in pairs(VK.Pages) do
		p.Visible = (k == key)
	end
	VK.CurrentPage = key
end

function VK.makeSlider(page, name, desc, min, max, default, callback)
	local card = VK.makeCard(page, 58)
	card:SetAttribute("vkName", string.lower(name .. " " .. desc))

	VK.new("TextLabel", {
		Size = UDim2.new(1, -70, 0, 15),
		Position = UDim2.new(0, 11, 0, 7),
		BackgroundTransparency = 1,
		Font = VK.FontBold,
		Text = name,
		TextSize = 11,
		TextColor3 = VK.Text,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, card)

	local valLbl = VK.new("TextLabel", {
		Size = UDim2.new(0, 54, 0, 15),
		Position = UDim2.new(1, -62, 0, 7),
		BackgroundTransparency = 1,
		Font = VK.FontBold,
		Text = tostring(default),
		TextSize = 10,
		TextColor3 = VK.Accent,
		TextXAlignment = Enum.TextXAlignment.Right,
	}, card)

	VK.new("TextLabel", {
		Size = UDim2.new(1, -22, 0, 12),
		Position = UDim2.new(0, 11, 0, 22),
		BackgroundTransparency = 1,
		Font = VK.Font,
		Text = desc,
		TextSize = 9,
		TextColor3 = VK.Dim,
		TextXAlignment = Enum.TextXAlignment.Left,
	}, card)

	local bar = VK.new("Frame", {
		Size = UDim2.new(1, -22, 0, 4),
		Position = UDim2.new(0, 11, 1, -12),
		BackgroundColor3 = VK.Border,
		BorderSizePixel = 0,
	}, card)
	VK.corner(bar, 2)

	local fill = VK.new("Frame", {
		Size = UDim2.new((default - min) / (max - min), 0, 1, 0),
		BackgroundColor3 = VK.Accent,
		BorderSizePixel = 0,
	}, bar)
	VK.corner(fill, 2)

	local dragBtn = VK.new("TextButton", {
		Size = UDim2.new(1, 0, 0, 20),
		Position = UDim2.new(0, 0, 1, -20),
		BackgroundTransparency = 1,
		Text = "",
	}, card)

	local ctrl = {value = default}
	local dragging = false

	local function apply(x)
		local rel = math.clamp((x - bar.AbsolutePosition.X) / bar.AbsoluteSize.X, 0, 1)
		local v = math.floor((min + (max - min) * rel) + 0.5)
		ctrl.value = v
		fill.Size = UDim2.new(rel, 0, 1, 0)
		valLbl.Text = tostring(v)
		if callback then VK.safe(function() callback(v) end) end
	end

	dragBtn.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			apply(i.Position.X)
		end
	end)
	UserInputService.InputChanged:Connect(function(i)
		if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
			apply(i.Position.X)
		end
	end)
	UserInputService.InputEnded:Connect(function() dragging = false end)

	VK.safe(function() if callback then callback(default) end end)
	return ctrl
end

function VK.setFly(state)
	VK.unbind("fly")
	if not state then return end
	local hrp = VK.hrp()
	if not hrp then return end
	local bv = Instance.new("BodyVelocity")
	bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
	bv.Velocity = Vector3.zero
	bv.Parent = hrp
	local bg = Instance.new("BodyGyro")
	bg.MaxTorque = Vector3.new(1e5, 1e5, 1e5)
	bg.P = 2500
	bg.D = 250
	bg.Parent = hrp

	VK.bind(RunService.RenderStepped:Connect(function()
		local cam = Workspace.CurrentCamera
		local dir = Vector3.zero
		local h = VK.Fly.held
		if h[Enum.KeyCode.W] then dir += cam.CFrame.LookVector end
		if h[Enum.KeyCode.S] then dir -= cam.CFrame.LookVector end
		if h[Enum.KeyCode.A] then dir -= cam.CFrame.RightVector end
		if h[Enum.KeyCode.D] then dir += cam.CFrame.RightVector end
		if h[Enum.KeyCode.Space] then dir += Vector3.new(0, 1, 0) end
		if h[Enum.KeyCode.LeftShift] then dir -= Vector3.new(0, 1, 0) end
		bv.Velocity = dir.Magnitude > 0 and dir.Unit * VK.Fly.speed or Vector3.zero
		bg.CFrame = cam.CFrame
	end), "fly")

	VK.bind(RunService.Stepped:Connect(function()
		if not VK.hrp() then VK.unbind("fly") end
	end), "fly")
end

VK.Aim = {mode = "Camera", fov = 140}
function VK.bestTarget()
	local cam = Workspace.CurrentCamera
	local hrp = VK.hrp()
	if not hrp then return nil end
	local best, bestD = nil, math.huge
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LP and VK.isAlive(plr) then
			local c = plr.Character
			local head = c and (c:FindFirstChild("Head") or c:FindFirstChild("HumanoidRootPart"))
			if head then
				local sp = cam:WorldToViewportPoint(head.Position)
				if sp.Z > 0 then
					local center = Vector2.new(cam.ViewportSize.X / 2, cam.ViewportSize.Y / 2)
					local d = (Vector2.new(sp.X, sp.Y) - center).Magnitude
					if d < VK.Aim.fov and d < bestD then
						bestD = d
						best = head
					end
				end
			end
		end
	end
	return best
end

function VK.setAim(state)
	pcall(function() RunService:UnbindFromRenderStep("VKAim") end)
	if not state then return end
	RunService:BindToRenderStep("VKAim", Enum.RenderPriority.Camera.Value - 1, function()
		local cam = Workspace.CurrentCamera
		local head = VK.bestTarget()
		if not head then return end
		if VK.Aim.mode == "Camera" then
			cam.CFrame = CFrame.lookAt(cam.CFrame.Position, head.Position)
		else
			local sp = cam:WorldToViewportPoint(head.Position)
			local m = UserInputService:GetMouseLocation()
			if mousemoverel then mousemoverel(sp.X - m.X, sp.Y - m.Y) end
		end
	end)
end

-- ---------------- TELEPORT ----------------
function VK.tpToName(q)
	q = string.lower(q or "")
	if q == "" then VK.notify("Nickname kosong") return end
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LP then
			local nm = string.lower(p.Name)
			local dn = string.lower(p.DisplayName)
			if string.find(nm, q, 1, true) or string.find(dn, q, 1, true) then
				local hrp = VK.hrp()
				local o = p.Character and p.Character:FindFirstChild("HumanoidRootPart")
				if hrp and o then
					hrp.CFrame = o.CFrame * CFrame.new(0, 0, 4)
					VK.notify("TP ke " .. p.Name)
				else
					VK.notify(p.Name .. " belum siap")
				end
				return
			end
		end
	end
	VK.notify("Nickname tidak ditemukan")
end

function VK.tpRandom()
	local pool = {}
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LP and VK.isAlive(p) then table.insert(pool, p) end
	end
	if #pool == 0 then VK.notify("Tidak ada pemain lain") return end
	local t = pool[math.random(#pool)]
	local hrp, o = VK.hrp(), t.Character and t.Character:FindFirstChild("HumanoidRootPart")
	if hrp and o then hrp.CFrame = o.CFrame * CFrame.new(0, 0, 4) end
end

	VK.new("UIListLayout", {Padding = UDim.new(0, 4), SortOrder = Enum.SortOrder.LayoutOrder}, list)

	function VK.refreshPlayers()
		for _, c in ipairs(list:GetChildren()) do
			if c:IsA("TextButton") then c:Destroy() end
		end
		local n = 0
		for _, plr in ipairs(Players:GetPlayers()) do
			if plr ~= LP then
				n += 1
				local row = VK.new("TextButton", {
					Size = UDim2.new(1, -4, 0, 22),
					BackgroundColor3 = VK.Bg,
					Font = VK.Font,
					Text = "  " .. plr.Name,
					TextSize = 10,
					TextColor3 = VK.Dim,
					TextXAlignment = Enum.TextXAlignment.Left,
					AutoButtonColor = false,
				}, list)
				row.LayoutOrder = n
				VK.corner(row, 5)
				VK.stroke(row, VK.Border)
				row.MouseButton1Click:Connect(function()
					local hrp = VK.hrp()
					local o = plr.Character and plr.Character:FindFirstChild("HumanoidRootPart")
					if hrp and o then
						hrp.CFrame = o.CFrame * CFrame.new(0, 0, 4)
						VK.notify("TP ke " .. plr.Name)
					end
				end)
			end
		end
		if n == 0 then
			VK.new("TextLabel", {
				Size = UDim2.new(1, -4, 0, 20),
				BackgroundTransparency = 1,
				Font = VK.Font,
				Text = "  tidak ada pemain lain",
				TextSize = 10,
				TextColor3 = VK.Dim,
				TextXAlignment = Enum.TextXAlignment.Left,
			}, list)
		end
	end
	VK.refreshPlayers()
	task.spawn(function()
		while gui.Parent do
			task.wait(2)
			if p.Visible then VK.refreshPlayers() end
		end
	end)
end

-- filter pencarian
searchInput:GetPropertyChangedSignal("Text"):Connect(function()
	local q = string.lower(searchInput.Text)
	local page = VK.Pages[VK.CurrentPage]
	if not page then return end
	for _, c in ipairs(page:GetChildren()) do
		if c:IsA("Frame") and c:GetAttribute("vkName") then
			local n = c:GetAttribute("vkName")
			c.Visible = (q == "" or string.find(n, q, 1, true) ~= nil)
		end
	end
end)
